def block(indices, b, name):
    fb, chb = phi(b); fb0 = fb(mp.mpf(0))
    C = []; r = []; chis = []
    for nn in indices:
        v, ch = testvec(nn, b); ce, co, nrm2 = coeffs(v)
        f, _ = phi(nn); r.append(f(mp.mpf(0)) / fb0); chis.append(ch)
        C.append(ce if name == "even" else co)
    Mb = E if name == "even" else O
    K = len(indices)
    Cm = mp.matrix(K, len(C[0]))
    for i in range(K):
        for j in range(len(C[0])): Cm[i, j] = C[i][j]
    M = Cm * Mb * Cm.T
    D = mp.matrix(K, K)
    for i in range(K):
        for j in range(K):
            D[i, j] = lam * ((1 - chis[i] if i == j else 0) + r[i] * r[j] * (1 - chb))
    out = []
    for k in range(1, K + 1):
        Dk = D[:k, :k]; Mk = M[:k, :k]
        Lc = mp.cholesky(Dk); Li = Lc ** -1
        A = Li * Mk * Li.T; A = (A + A.T) / 2
        ev = mp.eigsy(A, eigvals_only=True)
        out.append((k, min(ev), max(ev)))
    return out, [1 - c for c in chis]
