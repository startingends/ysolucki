"""e27 fourth experiment: heuristic constant from the kernel K_L against the measured one. usage: python3 kpred.py.txt L"""
import sys
src = open("/home/user/ysolucki/rh/experiments/e27_relative/vec.py.txt").read()
head = src.split("q, _ = weil.build")[0]
exec(head)
mp.mp.dps = 40
tail = src.split("# prolate functions on [-1, 1]")[1].split("def testvec")[0]
exec("# prolate functions on [-1, 1]" + tail)
def K(a):
    M = int(mp.floor(mp.exp(a) + mp.mpf(10) ** -30)); return sum((2 * Lm + a - 2 * mp.log(m)) / m for m in range(1, M + 1))
for nn, b, name in ((4, 0, "even"), (6, 2, "odd")):
    f, ch = phi(nn); fb, chb = phi(b); r = f(mp.mpf(0)) / fb(mp.mpf(0))
    G = lambda s: mp.sqrt(ch) * f(s) - r * mp.sqrt(chb) * fb(s)      # transform of the time-limited combination, up to a common factor
    S = 60
    brk = [mp.mpf(1)] + [mp.mpf(k) for k in range(2, S + 1)]
    num = mp.mpf(0); den = mp.mpf(0)
    gl = mp.calculus.quadrature.GaussLegendre(mp.mp).calc_nodes(4, mp.mp.prec)
    for a0, b0 in zip(brk[:-1], brk[1:]):
        h = (b0 - a0) / 2; m0 = (b0 + a0) / 2
        for x, w in gl:
            s = m0 + h * x; v = G(s) ** 2
            den += h * w * v; num += h * w * v * K(mp.log(s))
    lastv = sum(G(mp.mpf(S) - mp.mpf(j) / 8) ** 2 for j in range(8)) / 8 * S * S       # envelope ~ A / s^2
    den_t = lastv / S; num_t = lastv * (2 * Lm * (mp.log(S) + mp.euler) + mp.log(S)) / S   # crude tail (K ~ 2L(a + gamma) + a gamma)
    print("L=%s %s n=%d: leak (computed part) %s tail estimate %s ; kappa_pred = %s (without tail %s) ; 2L = %s ; mean log(xi/lambda) = see num" % (
        sys.argv[1], name, nn, mp.nstr(den, 5), mp.nstr(den_t, 3), mp.nstr((num + num_t) / (den + den_t), 6), mp.nstr(num / den, 6), mp.nstr(2 * Lm, 4)), flush=True)
