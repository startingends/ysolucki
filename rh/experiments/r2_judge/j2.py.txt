# ---- J7 Theorem F on random symmetric H (mixed signs), a > 0, s = Ha > 0
def theta_tensor(H, a, c4=0.25):
    m = len(a); s = H @ a; pi = a * s; K = H * a[None, :] / s[:, None]; K2 = K @ K
    br = lambda g, f, h: np.sum(pi * g * f * h)
    E = [np.eye(m)[i] / a[i] for i in range(m)]
    T = np.zeros((m, m, m))
    for i in range(m):
        for j in range(m):
            for k in range(m):
                g, f, h = E[i], E[j], E[k]
                T[i, j, k] = 0.5 * (br(g, K @ f, K @ h) + br(f, K @ g, K @ h) + br(h, K @ g, K @ f)) - 0.25 * (br(K2 @ g, f, h) + br(g, K2 @ f, h) + br(g, f, K2 @ h)) + c4 * br(g, f, h)
    return T
def F5(H, a):
    m = len(a); s = H @ a; HDH = H @ np.diag(a / s) @ H; T = np.zeros((m, m, m))
    for i in range(m):
        for j in range(m):
            for k in range(m):
                if len({i, j, k}) == 3:
                    T[i, j, k] = 0.5 * (H[i, j] * H[i, k] / s[i] + H[i, j] * H[j, k] / s[j] + H[i, k] * H[j, k] / s[k])
                elif i == j == k:
                    T[i, i, i] = 1.5 * H[i, i] ** 2 / s[i] - 0.75 * HDH[i, i] / a[i] + s[i] / (4 * a[i] ** 2)
                else:
                    r = i if (i == j or i == k) else j           # repeated index
                    o = [x for x in (i, j, k) if x != r][0]
                    T[i, j, k] = H[r, r] * H[r, o] / s[r] + H[r, o] ** 2 / (2 * s[o]) - HDH[r, o] / (4 * a[r])
    return T
def gamma2_slices(H, a):
    """slice k from the definition: T_k = l_k l_k^T/s_k - s_k * A^{-1} (Gamma_2 - Gamma)_k A^{-1}, Gamma's by definition with L = K - I"""
    m = len(a); s = H @ a; K = H * a[None, :] / s[:, None]; Lg = K - np.eye(m); Ai = np.diag(1 / a)
    def Gam(k):   # matrix of f -> Gamma(f)(k) = (1/2)[L(f^2)(k) - 2 f_k (Lf)_k]
        G = 0.5 * np.diag(Lg[k]); G[k, :] -= 0.5 * Lg[k]; G[:, k] -= 0.5 * Lg[k]; return G
    Gs = [Gam(k) for k in range(m)]
    out = np.zeros((m, m, m))
    for k in range(m):
        G2 = 0.5 * sum(Lg[k, y] * Gs[y] for y in range(m)) - 0.5 * (Gs[k] @ Lg + Lg.T @ Gs[k])
        out[:, :, k] = np.outer(H[:, k], H[:, k]) / s[k] - s[k] * Ai @ (G2 - Gs[k]) @ Ai
    return out

# ---- J8 Lemma C
def cone(H, a):
    l = H @ a; s = a @ l
    return (np.einsum('ij,k->ijk', H, l) + np.einsum('ik,j->ijk', H, l) + np.einsum('jk,i->ijk', H, l) - (2 / s) * np.einsum('i,j,k->ijk', l, l, l)) / s

# ---- J9 CD(rho, n) on the resampling chain (uniform, m points): best rho with Gamma_2 >= rho Gamma + (1/n)(Lf)^2 at every site
import scipy.linalg as sl
def cd(m, n):
    K = np.ones((m, m)) / m; Lg = K - np.eye(m)
    def Gam(k):
        G = 0.5 * np.diag(Lg[k]); G[k, :] -= 0.5 * Lg[k]; G[:, k] -= 0.5 * Lg[k]; return G
    Gs = [Gam(k) for k in range(m)]; k = 0
    G2 = 0.5 * sum(Lg[k, y] * Gs[y] for y in range(m)) - 0.5 * (Gs[k] @ Lg + Lg.T @ Gs[k])
    A = G2 - np.outer(Lg[k], Lg[k]) / n; idx = list(range(1, m))
    return sl.eigh(A[np.ix_(idx, idx)], Gs[k][np.ix_(idx, idx)], eigvals_only=True)[0]
