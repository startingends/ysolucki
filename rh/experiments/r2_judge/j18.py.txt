def posneg(M, thr=mp.mpf('1e-90')):
    ev = mp.eigsy((M + M.T) / 2, eigvals_only=True)
    return sum(1 for e in ev if e > thr), sum(1 for e in ev if e < -thr)
def cvec(L, N, lat):
    L = mp.mpf(L); w = wb.freqs(L, N, lat); ce = []; so = []
    for om in w:
        I = 2 * mp.sinh(mp.mpc(mp.mpf(1) / 2, om) * L) / mp.mpc(mp.mpf(1) / 2, om)
        nrm = mp.sqrt(2 * L) if om == 0 else mp.sqrt(L)
        ce.append(I.real / nrm); so.append(I.imag / nrm)
    if lat == 'int': so = so[1:]
    return mp.matrix(ce), mp.matrix(so)
