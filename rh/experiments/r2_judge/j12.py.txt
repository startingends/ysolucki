import sys, math
exec(open("/home/user/ysolucki/rh/experiments/r2_judge/j1.py.txt").read().split('which = sys.argv[1]')[0].replace("dps = 40", "dps = 90"))
L, N = 1.0, 40
A = add(arch(L, N, pole=True), atoms(L, N, lam(math.exp(2 * L))))
U = atoms(L, N, {6: arb(1)})
print("lowest Weil eigenvalue (even, odd):", [mp.nstr(e, 6) for e in low(A)])
lo_, hi_ = [], []
for b in (0, 1):
    C = mp.cholesky(A[b]); Ci = mp.inverse(C); G = Ci * U[b] * Ci.T
    G = (G + G.T) / 2
    ev = mp.eigsy(G, eigvals_only=True); gmin, gmax = min(ev), max(ev)
    lo_.append(-1 / gmax); hi_.append(-1 / gmin)
    print("block %d: feasible delta_6 in [%s, %s]" % (b, mp.nstr(-1 / gmax, 5), mp.nstr(-1 / gmin, 5)))
print("J12 feasible interval for c_6 perturbation: [%s, %s]" % (mp.nstr(max(lo_), 5), mp.nstr(min(hi_), 5)))
def beurling(X):
    out = {}
    for p in (2.1, 3, 5, 7):
        pk = p
        while pk <= X:
            out[pk] = arb(p).log(); pk *= p
    return out
mp.mp.dps = 30
Ab = add(arch(L, N, pole=True), atoms(L, N, beurling(math.exp(2 * L))))
best = max((min(low(add(Ab, U, t))), t) for t in [x / 20 for x in range(-40, 41)])
print("J12 control Beurling base: lowest eig at delta=0: %s ; best over delta_6 in [-2,2] step 0.05: %s at delta=%s" % (mp.nstr(min(low(Ab)), 5), mp.nstr(best[0], 5), best[1]))
