def chain(N):
    ev, V = np.linalg.eigh(N); lam1 = ev[-1]; u = np.abs(V[:, -1])
    q = N * u[None, :] / u[:, None]; np.fill_diagonal(q, 0.0)
    Lg = q - np.diag(q.sum(1)); return ev, lam1, u, q, Lg
def gam_mats(q, Lg):
    M = len(q); Gs = []
    for i in range(M):
        D = np.zeros((M, M))
        # Gamma f(i) = (1/2) sum_y q_iy (f_y - f_i)^2  as a quadratic form
        D[np.diag_indices(M)] = 0.5 * q[i]
        D[i, :] -= 0.5 * q[i]; D[:, i] -= 0.5 * q[i]; D[i, i] += 0.5 * q[i].sum()
        Gs.append(D)
    G2 = []
    for i in range(M):
        LG = sum(q[i, y] * (Gs[y] - Gs[i]) for y in range(M) if y != i)
        G2.append(0.5 * LG - 0.5 * (Gs[i] @ Lg + Lg.T @ Gs[i]))
    return Gs, G2
def be_const(A, B, i):
    idx = [k for k in range(len(A)) if k != i]
    return sl.eigh(A[np.ix_(idx, idx)], B[np.ix_(idx, idx)], eigvals_only=True)[0]
