import sys, math, numpy as np, mpmath as mp
mp.mp.dps = 25
L = float(sys.argv[1]); M = int(sys.argv[2]); d = 2 * L / M
w = lambda u: mp.exp(-u / 2) / (-mp.expm1(-2 * u))
tri = lambda t: max(0.0, 1 - abs(t))
def Wk(k):
    f = lambda u: w(u) * (1 - abs(u - k * d) / d)
    return mp.quad(f, [0, d, 2 * d]) if k == 1 else mp.quad(f, [(k - 1) * d, k * d, (k + 1) * d])
diag = float(-(-mp.log(mp.pi) + mp.quad(lambda u: mp.exp(-2 * u) / u - 2 * w(u) * (1 - u / d), [0, d / 2, d]) + mp.e1(2 * d)))
ta = np.array([0.0] + [float(Wk(k)) for k in range(1, M)])
idx = np.abs(np.subtract.outer(np.arange(M), np.arange(M))); E = np.eye(M)
Bc = np.linalg.qr(np.hstack([np.ones((M, 1)), np.eye(M)[:, 1:]]))[0][:, 1:]
out = {}
for kind in ("zeta", "beurling"):
    lam = {}
    for p in ((2.1 if kind == 'beurling' else 2), 3, 5, 7):
        pk = p
        while pk <= math.exp(2 * L): lam[pk] = math.log(p); pk *= p
    tp = np.array([sum(c / math.sqrt(n) * (tri((math.log(n) - k * d) / d) + tri((math.log(n) + k * d) / d)) for n, c in lam.items()) for k in range(M)])
    N = (ta + tp)[idx]; np.fill_diagonal(N, diag)
    ev, V = np.linalg.eigh(N); lam1 = ev[-1]; u = np.abs(V[:, -1]); pi = u ** 2
    q = N * u[None, :] / u[:, None]; np.fill_diagonal(q, 0); Lg = q - np.diag(q.sum(1))
    Gs = np.array([sum(0.5 * q[i, y] * np.outer(E[y] - E[i], E[y] - E[i]) for y in range(M) if y != i) for i in range(M)])
    LG = np.einsum('xy,yab->xab', Lg, Gs)
    cnt = []; tot = np.zeros((M, M))
    for i in range(M):
        D = 0.5 * (LG[i] - (Gs[i] @ Lg + Lg.T @ Gs[i])) - lam1 * Gs[i]
        tot += pi[i] * D
        e = np.linalg.eigvalsh(Bc.T @ D @ Bc)
        cnt.append(int((e < -1e-9).sum()))
    et = np.linalg.eigvalsh(Bc.T @ tot @ Bc)
    out[kind] = cnt
    print("%-8s L=%g M=%d: negative directions of Gamma_2 - lambda_1 Gamma per site: %s" % (kind, L, M, cnt))
    print("          sites with exactly one: %d of %d ; integrated form: %d negative directions (min eig %.3e)" % (cnt.count(1), M, (et < -1e-9).sum(), et[0]))
print("sites where zeta and Beurling counts agree: %d of %d" % (sum(a == b for a, b in zip(out['zeta'], out['beurling'])), M))
