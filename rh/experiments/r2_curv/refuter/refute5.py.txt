def blocks(N, c, s):
    M = N.shape[0]; h = M//2
    Be = np.zeros((M, h)); Bo = np.zeros((M, h))
    for i in range(h):
        Be[i, i] = Be[M-1-i, i] = 1/math.sqrt(2)
        Bo[i, i] = -1/math.sqrt(2); Bo[M-1-i, i] = 1/math.sqrt(2)
    return Be.T@N@Be, Bo.T@N@Bo, Be.T@c, Bo.T@s

def gen_build(L, M, alpha, atoms):
    wf = lambda u: mp.exp(-alpha*u)/(-mp.expm1(-2*u))
    d = mp.mpf(2*L)/M; k = np.zeros(M)
    for m in range(1, M):
        a = (m-1)*d; b = m*d; c = (m+1)*d
        k[m] = float(mp.quad(lambda u: wf(u)*(u-a)/d, [a, b]) + mp.quad(lambda u: wf(u)*(c-u)/d, [b, c]))
    df = float(d); karch = k.copy()
    for (pos, wt) in atoms:
        for m in range(1, M):
            t = 1 - abs(pos - m*df)/df
            if t > 0: k[m] += wt*t
    return k, karch
def negfrac(k):
    M = len(k); tot = M*(M-1); neg = sum(2*(M-m) for m in range(1, M) if k[m] < 0)
    return neg/tot, k[1:].min(), [m for m in range(1, M) if k[m] < 0]
def logder(a, X):
    # a[n] for n>=1 with a[1]=1 ; returns Lam[n] with a_n log n = sum_{d|n} Lam(d) a(n/d)
    Lam = {}
    for n in range(2, X+1):
        v = a(n)*math.log(n)
        for dd in range(2, n):
            if n % dd == 0: v -= Lam[dd]*a(n//dd)
        Lam[n] = v
    return Lam

print("R12: dial theta^n")
for nn in (1.0, 1.4, 2.0):
    X = 54
    # theta^n coefficients via J.C.P. Miller recurrence on theta = 1 + 2 sum q^{k^2}
    th = [0.0]*(X+1); th[0] = 1.0
    kq = 1
    while kq*kq <= X: th[kq*kq] = 2.0; kq += 1
    r = [0.0]*(X+1); r[0] = 1.0
    for m in range(1, X+1):
        r[m] = sum((nn*j - (m - j))*th[j]*r[m-j] for j in range(1, m+1))/m
    a = lambda m: r[m]/(2*nn)
    assert abs(a(1) - 1) < 1e-12
    c = logder(a, X)
    atoms = [(0.5*math.log(m), 0.5*v*m**(-nn/4)) for m, v in c.items() if 0.5*math.log(m) <= 2.0]
    k, ka = gen_build(1.0, 40, nn/2, atoms)
    f, mn, where = negfrac(k)
    print(f"  n={nn}: c_m (m<=12) {[round(c[m],4) for m in range(2,13)]}  negative fraction {100*f:.2f}%  min entry {mn:.4f} at offsets {where}")
