def mangoldt(m):
    for p in range(2,m+1):
        if m%p==0:
            k=m
            while k%p==0: k//=p
            return math.log(p) if k==1 else 0.0
    return 0.0
for lo in (2,1):

    e=np.linalg.eigvalsh(A); print(f"D1 a,b>={lo}: min eig {e[0]:.3e}, rank {int((abs(e)>1e-10).sum())}")

def parts(x):
    S=np.zeros((M,M)); Hs=np.zeros((M,M)); O=np.zeros((M,M))
    def e(i):
        v=np.zeros(M); v[i]=1; return v
    for a in range(-M+1,M):
        if a==0 or not (0<=x+a<M): continue
        for b in range(-M+1,M):
            if b==0: continue
            ka=kk[abs(a)]; kb=kk[abs(b)]
            inb=0<=x+b<M; inab=0<=x+a+b<M
            A=e(x+a)-e(x)
            if inb:
                B=e(x+b)-e(x)
                rho=u[x+a+b]/u[x] if inab else 0.0
                Hab=u[x+a]*u[x+b]/u[x]**2-rho
                Hs+=0.5*ka*kb*Hab*0.5*(np.outer(A,B)+np.outer(B,A))
                if inab:
                    D=e(x+a+b)-e(x+a)-e(x+b)+e(x)
                    S+=0.25*ka*kb*rho*np.outer(D,D)
            elif inab:
                rho=u[x+a+b]/u[x]
                v1=e(x+a+b)-e(x+a); v2=e(x+a+b)-e(x)
                O+=0.5*ka*kb*rho*(np.outer(v1,v1)-0.5*np.outer(v2,v2))
    return S,Hs,O
for x in (M//2,0,5):
    S,Hs,O=parts(x)
    err=np.abs(S+Hs+O-G2[x]).max()
    keep=[i for i in range(M) if i!=x]
    dg=np.sqrt(np.diag(G[x])[keep])
    def nm(Z): return np.linalg.eigvalsh(Z[np.ix_(keep,keep)]/np.outer(dg,dg))
    print(f"B3 site {x}: |S+H+O-Gamma2|max={err:.1e}; raw min eig S {np.linalg.eigvalsh(S)[0]:.3f}, H {np.linalg.eigvalsh(Hs)[0]:.3f} (max {np.linalg.eigvalsh(Hs)[-1]:.2f}), O {np.linalg.eigvalsh(O)[0]:.3f};"
          f" Gamma-normalised/lam1: S {nm(S)[0]/lam1:.3f}, S+H {nm(S+Hs)[0]/lam1:.3f}, S+O {nm(S+O)[0]/lam1:.3f}, all {nm(S+Hs+O)[0]/lam1:.3f}")
