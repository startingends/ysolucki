def arch(t, V, alpha, c0):
    # c0 + int_0^inf [e^-u/u - e^{-alpha u}(1-u/2V)_+ cos(tu/2)/(1-e^-u)] du
    f1 = lambda u: (math.exp(-u)/u - math.exp(-alpha*u)*(1-u/(2*V))*math.cos(t*u/2)/(-math.expm1(-u))) if u > 1e-12 else (alpha - 0.5 + 1/(2*V))
    a, _ = quad(f1, 0, 2*V, limit=2000, epsabs=1e-12, epsrel=1e-12)
    from scipy.special import exp1
    return c0 + a + exp1(2*V)
def pole(t, V):
    # h(i/2)+h(-i/2), h(r)=int g e^{irx}, g=(1-|x|/V)e^{-itx}
    tot = 0
    for a in (0.5 - 1j*t, -0.5 - 1j*t, 0.5 + 1j*t, -0.5 + 1j*t):   # int_0^V (1-x/V) e^{a x} dx
        tot += (np.exp(a*V) - 1 - a*V)/(a*a*V)
    return tot.real
def lam_zeta(X):
    out = []
    for p in primerange(2, int(X)+1):
        q = p
        while q <= X: out.append((q, math.log(p))); q *= p
    return out
def lam_dh(N):
    kap = (math.sqrt(10 - 2*math.sqrt(5)) - 2)/(math.sqrt(5) - 1)
    a = np.array([[0.0, 1.0, kap, -kap, -1.0][n % 5] for n in range(N+1)])
    L = a*np.log(np.maximum(np.arange(N+1), 1))
    for d in range(2, N//2 + 1):          # L[d] final here
        if L[d] != 0:
            m = np.arange(2, N//d + 1); L[d*m] -= L[d]*a[m]
    return [(n, L[n]) for n in range(2, N+1) if abs(L[n]) > 1e-15]
def S_prime(mode, t, X, lam):
    V = math.log(X)
    s = sum(L*(1 - math.log(n)/V)*n**-0.5*math.cos(t*math.log(n)) for n, L in lam if n <= X)
    if mode == "zeta": return pole(t, V) + arch(t, V, 0.25, -math.log(math.pi)) - 2*s
    return arch(t, V, 0.75, math.log(5/math.pi)) - 2*s
def K(z, V):  # (4/V) sin^2(zV/2)/z^2, complex z ok
    return (4/V)*np.sin(z*V/2)**2/z**2
