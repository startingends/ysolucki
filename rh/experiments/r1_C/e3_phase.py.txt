"""E3: monotone-phase test.  D_X(t) = d/dt [theta(t) + Im log P_X(1/2+it) + pole correction]
 = theta'(t) - sum_{n<=X} Lambda(n) (1 - log n/log X) n^{-1/2} cos(t log n) + pole term.
By the explicit formula D_X(t) = sum over zeros of a Fejer kernel (>= 0 termwise iff the zero is on the line).
Report min over t in [0.5, TMAX] for zeta and for Davenport-Heilbronn (formal Lambda_f).
Derivative taken numerically from the same phase used in E1 (grid step 0.002).
usage: python3 e3_phase.py.txt <zeta|dh> TMAX X1 X2 ...
"""
import sys, math, numpy as np
from scipy.special import loggamma, exp1
mode, TMAX = sys.argv[1], float(sys.argv[2]); Xs = [float(x) for x in sys.argv[3:]]
T = np.linspace(0.5, TMAX, int((TMAX - 0.5) * 500) + 1)
def lam_zeta(N):
    s = np.ones(N + 1, bool); s[:2] = False
    for i in range(2, int(N ** 0.5) + 1):
        if s[i]: s[i * i::i] = False
    out = []
    for p in np.nonzero(s)[0]:
        q = int(p)
        while q <= N: out.append((float(q), math.log(p))); q *= int(p)
    return out
def lam_dh(N):
    kap = (math.sqrt(10 - 2 * math.sqrt(5)) - 2) / (math.sqrt(5) - 1)
    a = [0.0] + [[0.0, 1.0, kap, -kap, -1.0][n % 5] for n in range(1, N + 1)]
    L = [0.0] * (N + 1)
    divs = [[] for _ in range(N + 1)]
    for d in range(2, N + 1):
        for m in range(2 * d, N + 1, d): divs[m].append(d)
    for n in range(2, N + 1):
        L[n] = a[n] * math.log(n) - sum(L[d] * a[n // d] for d in divs[n])
    return [(float(n), L[n]) for n in range(2, N + 1) if abs(L[n]) > 1e-14]
lam = (lam_dh if mode == "dh" else lam_zeta)(int(max(Xs)))
for X in Xs:
    V = math.log(X)
    D = np.zeros_like(T)
    for n, L in lam:
        if n > X: break
        ln = math.log(n)
        D -= L * (1 - ln / V) * n ** -0.5 * np.cos(T * ln)
    if mode == "dh":
        th = np.imag(loggamma(0.75 + 0.5j * T)) + 0.5 * T * math.log(5 / math.pi); corr = 0 * T
    else:
        th = np.imag(loggamma(0.25 + 0.5j * T)) - 0.5 * T * math.log(math.pi)
        mz = (-0.5 + 1j * T) * V
        G = -exp1(mz) - np.log(mz) - np.euler_gamma - (np.exp(-mz) - 1) / (-mz) + 1
        corr = -np.imag(G) + np.arctan(2 * T)
    D += np.gradient(th + corr, T)
    i = int(np.argmin(D))
    print("%s X=%-8g min_t D_X(t) = %+.5f at t = %.3f ;  min * log X = %+.4f ; fraction of t with D<0: %.4f"
          % (mode, X, D[i], T[i], D[i] * V, float(np.mean(D < 0))))
