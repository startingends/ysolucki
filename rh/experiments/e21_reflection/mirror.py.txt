import math, sys, importlib.util, importlib.machinery
import numpy as np
from sympy import factorint
P = "/home/user/ysolucki/rh/experiments/r1_A/weil.py.txt"
spec = importlib.util.spec_from_file_location("weil", P, loader=importlib.machinery.SourceFileLoader("weil", P))
weil = importlib.util.module_from_spec(spec); spec.loader.exec_module(weil)
A, X = 12, 144

def mangoldt(n):
    f = factorint(n)
    return math.log(list(f)[0]) if len(f) == 1 else 0.0
def chi4(n): return 0 if n % 2 == 0 else (1 if n % 4 == 1 else -1)
def logderiv(a):                       # a[1] = 1; returns c with a_m log m = sum_{d|m} c_d a_{m/d}
    c = [0.0] * (X + 1)
    for m in range(2, X + 1):
        s = a[m] * math.log(m)
        for d in range(2, m):
            if m % d == 0: s -= c[d] * a[m // d]
        c[m] = s
    return c
def dial(n):                           # Dirichlet coefficients r_n(m)/(2n) of theta^n, theta = 1 + 2 sum q^(k^2)
    f = [0.0] * (X + 1); f[0] = 1.0
    k = 1
    while k * k <= X: f[k * k] = 2.0; k += 1
    b = [0.0] * (X + 1); b[0] = 1.0
    for m in range(1, X + 1):
        b[m] = sum((n * j - (m - j)) * f[j] * b[m - j] for j in range(1, m + 1)) / m
    return [0.0] + [b[m] / (2 * n) for m in range(1, X + 1)]

cases = {}
cases["zeta"] = [0, 0] + [mangoldt(n) for n in range(2, X + 1)]
dh = weil.dh_logderiv_coeffs(X); cases["impostor (Davenport-Heilbronn)"] = [0, 0] + [float(dh[n].mid()) for n in range(2, X + 1)]
for n in (1.0, 1.2, 1.4, 1.8, 2.0, 3.0, 4.0): cases["theta dial n=%.1f" % n] = logderiv(dial(n))
cases["L(chi_4) alone"] = [0, 0] + [mangoldt(n) * chi4(n) for n in range(2, X + 1)]
cases["zeta * L(chi_4)  (check of dial n=2)"] = [0, 0] + [mangoldt(n) * (1 + chi4(n)) for n in range(2, X + 1)]
cases["zeta(2s)          (check of dial n=1)"] = [0, 0] + [2 * mangoldt(int(round(math.sqrt(n)))) if int(round(math.sqrt(n))) ** 2 == n else 0.0 for n in range(2, X + 1)]
print("%-40s %12s %6s %10s   %s" % ("function", "min eig", "rank", "#neg eig", "c_2..c_9"))
for name, c in cases.items():
    M = np.array([[c[a * b] for b in range(2, A + 1)] for a in range(2, A + 1)])
    ev = np.linalg.eigvalsh(M)
    print("%-40s %12.3e %6d %10d   %s" % (name, ev[0], int(np.sum(np.abs(ev) > 1e-9)), int(np.sum(ev < -1e-9)),
          " ".join("%.3f" % c[k] for k in range(2, 10))))
d2 = np.array(cases["theta dial n=2.0"]) - np.array(cases["zeta * L(chi_4)  (check of dial n=2)"])
d1 = np.array(cases["theta dial n=1.0"]) - np.array(cases["zeta(2s)          (check of dial n=1)"])
print("check: max |dial(2) - zeta*L(chi_4)| = %.2e ; max |dial(1) - zeta(2s)| = %.2e" % (np.abs(d2).max(), np.abs(d1).max()))
