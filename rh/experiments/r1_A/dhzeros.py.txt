"""Zeros of the Davenport-Heilbronn function up to height T (on-line by sign changes of Lambda(1/2+it), off-line by findroot).
f(s) = 5^{-s} [ zeta(s,1/5) + xi zeta(s,2/5) - xi zeta(s,3/5) - zeta(s,4/5) ],  Lambda(s) = (5/pi)^{s/2} Gamma((s+1)/2) f(s) = Lambda(1-s)."""
import mpmath as mp, sys, json
mp.mp.dps = 25
T = float(sys.argv[1])
xi = (mp.sqrt(10 - 2 * mp.sqrt(5)) - 2) / (mp.sqrt(5) - 1)
def f(s):
    return 5 ** (-s) * (mp.zeta(s, mp.mpf(1) / 5) + xi * mp.zeta(s, mp.mpf(2) / 5) - xi * mp.zeta(s, mp.mpf(3) / 5) - mp.zeta(s, mp.mpf(4) / 5))
def Lam(s):
    return mp.exp(mp.log(5 / mp.pi) * s / 2 + mp.loggamma((s + 1) / 2)) * f(s)
def Z(t):   # real on the line; rescaled to O(1)
    s = mp.mpc(0.5, t)
    v = Lam(s) * mp.exp(mp.pi * t / 4)
    return v.real, abs(v.imag)
# functional equation check
s0 = mp.mpc(0.3, 7.7)
print("# FE check |Lam(s)-Lam(1-s)|/|Lam(s)| =", mp.nstr(abs(Lam(s0) - Lam(1 - s0)) / abs(Lam(s0)), 5), " max|Im Z| sample:", mp.nstr(Z(33.3)[1], 5))
h = 0.02
ts = [mp.mpf(0.005) + h * i for i in range(int(T / h) + 1)]
vals = [Z(t)[0] for t in ts]
online = []
for i in range(len(ts) - 1):
    if vals[i] * vals[i + 1] < 0:
        online.append(mp.findroot(lambda t: Z(t)[0], (ts[i], ts[i + 1]), solver='anderson'))
# expected count from the argument of the gamma factor (main term), N(T) = theta(T)/pi + S(T), theta(T) = arg[(5/pi)^{iT/2} Gamma(3/4 + iT/2)]
theta = lambda t: (mp.log(5 / mp.pi) * t / 2 + mp.im(mp.loggamma(mp.mpc(0.75, t / 2))))
print(f"# on-line zeros in (0,{T}): {len(online)} ; main term theta(T)/pi = {mp.nstr(theta(T)/mp.pi, 8)}")
off = []
for guess in [mp.mpc(0.808517, 85.699348), mp.mpc(0.650830, 114.163343), mp.mpc(0.574356, 166.479306), mp.mpc(0.724258, 176.702461)]:
    if guess.imag < T:
        try:
            r = mp.findroot(f, guess)
            off.append(r); print("# off-line zero:", mp.nstr(r, 20), " |f| =", mp.nstr(abs(f(r)), 5))
        except Exception as e:
            print("# findroot failed for", guess, e)
print(f"# total count: on-line {len(online)} + 2*off-line {2*len(off)} = {len(online) + 2*len(off)}")
json.dump({"online": [str(t) for t in online], "off": [[str(r.real), str(r.imag)] for r in off], "T": T}, open("/home/user/ysolucki/rh/experiments/r1_A/dh_zeros.json", "w"))
