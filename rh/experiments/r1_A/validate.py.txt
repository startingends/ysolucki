gam = rhlib.zeros(5000)
def zero_side(coef, L):
    # coef: dict k -> real coefficient in exp basis ; Q = sum over +-gamma |fhat(gamma)|^2
    tot = 0.0
    for g in gam:
        for sgn in (1, -1):
            r = sgn * g
            x = r * L
            s = 0.0
            for k, c in coef.items():
                s += c * (-1) ** k / (x + math.pi * k)
            v = math.sqrt(2 * L) * math.sin(x) * s
            tot += v * v
    return tot

tests = {
