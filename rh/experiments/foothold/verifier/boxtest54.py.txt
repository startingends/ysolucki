def Gblocks(y):
    g = Pb.gamma(mp.mpf(y)); out = []
    for idx in (range(0, NB, 2), range(1, NB, 2)):
        idx = list(idx); B = mp.matrix(len(idx), len(idx))

        out.append(B)
    return out
