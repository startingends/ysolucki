"""V-FH recomputation (ii): Theorem B at L = 1/2, points just outside / inside the claimed box.
Route P: my own polynomial code (degree < 28: exactly the hypothesis of Theorem B (ii)).
Route F: the project's periodic Fourier/Arb builder at N = 160 and 320 (a different basis; only an upper bound for the
infimum over L^2, so a positive Fourier value does NOT contradict infeasibility, a negative one confirms it).
lam(y, m) = min(lowest even, lowest odd eigenvalue);  lam*(y) = max over m >= 0.   usage: boxtest.py.txt"""

t0 = time.time()
mp.mp.dps = 34; flint.ctx.prec = 160
L = mp.mpf(1) / 2
Pb = Poly(L, 28)
A = Pb.blocks()
def Gblocks(y):
    g = Pb.gamma(mp.mpf(y)); out = []
    for idx in (range(0, 28, 2), range(1, 28, 2)):
        idx = list(idx); B = mp.matrix(len(idx), len(idx))
        for a_, j in enumerate(idx):
            for b_, k in enumerate(idx): B[a_, b_] = (g[j, k] + g[k, j]) / 2
        out.append(B)
    return out
def lamP(G, m):
    return min(min(mp.eigsy(A[b] - 2 * mp.mpf(m) * G[b], eigvals_only=True)) for b in (0, 1))
def gold(f, a, b, it=80):
    gr = (mp.sqrt(5) - 1) / 2
    c = b - gr * (b - a); d = a + gr * (b - a); fc = f(c); fd = f(d)
    for _ in range(it):
        if fc > fd: b, d, fd = d, c, fc; c = b - gr * (b - a); fc = f(c)
        else: a, c, fc = c, d, fd; d = a + gr * (b - a); fd = f(d)
    x = (a + b) / 2
    return x, f(x)
def starP(y):
    G = Gblocks(y)
    return gold(lambda m: lamP(G, m), mp.mpf(0), mp.mpf(6))
FA = {N: f_arch(arb(1) / 2, N) for N in (160, 320)}
def starF(y, N):
    B = f_atom(arb(1) / 2, N, arb(mp.nstr(y, 40)))
    f = lambda m: min(lows(FA[N], B, float(m)))
    m, v = gold(f, mp.mpf(0), mp.mpf(6), it=60)
    return float(m), v, B

print("--- m direction: max over y of lam(y, m)  (P: grid on (0,1) then refinement; F at the P-argmax and neighbours) ---")
def best_y_P(m):
    # coarse grid over (0, 1), then fine grid around the best
    best = None
    for i in range(1, 100):
        y = mp.mpf(i) / 100
        v = lamP(Gblocks(y), m)
        if best is None or v > best[1]: best = (y, v)
    lo, hi = best[0] - mp.mpf('0.01'), best[0] + mp.mpf('0.01')
    for rnd in range(6):
        pts = [lo + (hi - lo) * i / 20 for i in range(21)]
        vals = [(y, lamP(Gblocks(y), m)) for y in pts if 0 < y < 1]
        best = max(vals, key=lambda t: t[1])
        w = (hi - lo) / 10
        lo, hi = best[0] - w, best[0] + w
    return best
