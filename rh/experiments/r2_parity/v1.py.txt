"""validation of implementation B: (1) closed form vs kernel quadrature, (2) entrywise vs r1_A builder on the integer
lattice, (3) Q(f) vs sum over zeros."""
import sys, time, importlib.util, importlib.machinery
import mpmath as mp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m
wb = load("wb", "/home/user/ysolucki/rh/experiments/r2_parity/wb.py.txt")
mp.mp.dps = 50
t0 = time.time()
print("(1) Phi_arch: closed form vs direct quadrature of the kernel, dps 50")
for L, om in ((1.0, 0.0), (1.0, 4.71238898038469), (1.0, 51.3), (0.5, 9.42477796076938), (0.4581, 30.0)):
    a = wb.phi_arch(mp.mpf(om), mp.mpf(L)); b = wb.phi_arch_quad(mp.mpf(om), mp.mpf(L))
    print("  L=%g om=%g  Phi=%s  |diff Phi|=%s  |diff ImPhi'|=%s" % (L, om, mp.nstr(a[0], 12), mp.nstr(abs(a[0] - b[0]), 3),
          mp.nstr(abs(a[1].imag - b[1].imag), 3)))
print("  %.1fs" % (time.time() - t0))

print("(2) entrywise vs r1_A builder (integer lattice), L=1, N=12, full zeta form")
import flint
flint.ctx.prec = 300
weil = load("weil", "/home/user/ysolucki/rh/experiments/r1_A/weil.py.txt")
L, N = 1.0, 12
q, _ = weil.build(L, N, kind='zeta', pole=True, primes=True, arch=True)
EA = weil.even_block(q, N); OA = weil.odd_block(q, N)
def tomp(M):
    R = mp.matrix(M.nrows(), M.ncols())
    for i in range(M.nrows()):
        for j in range(M.ncols()):
            R[i, j] = mp.mpf(M[i, j].mid().str(60, radius=False))
    return R
EA, OA = tomp(EA), tomp(OA)
def full(L, N, lat, atoms):
    M = wb.add(wb.arch_blocks(L, N, lat), wb.pole_blocks(L, N, lat))
    for x, c in atoms.items():
        M = wb.add(M, wb.atom_blocks(L, N, lat, x), c)
    return M
lam = {n: v for n, v in wb.mangoldt(7).items()}
EB, OB = full(L, N, 'int', lam)
de = max(abs(EA[i, j] - EB[i, j]) for i in range(N + 1) for j in range(N + 1))
do = max(abs(OA[i, j] - OB[i, j]) for i in range(N) for j in range(N))
print("  max |A - B| even block %s ; odd block %s" % (mp.nstr(de, 3), mp.nstr(do, 3)))
bl = dict(lam); del bl[2]; del bl[4]; bl[mp.mpf('2.1')] = mp.log(mp.mpf('2.1')); bl[mp.mpf('4.41')] = mp.log(mp.mpf('2.1'))
EC, OC = full(L, N, 'int', bl)
print("  control (2 -> 2.1 in B): max diff even %s" % mp.nstr(max(abs(EA[i, j] - EC[i, j]) for i in range(N + 1) for j in range(N + 1)), 3))

print("(3) Q(f) vs sum over zeros; f = w1 cos(w0 x) + w0 cos(w1 x) on [-1,1], half lattice (f(L)=f'(L)=0)")
sys.path.insert(0, "/home/user/ysolucki/tools"); import rhlib
L = mp.mpf(1); w = wb.freqs(L, 2, 'half')
E, O = full(L, 2, 'half', lam)
v = mp.matrix([w[1] * mp.sqrt(L), w[0] * mp.sqrt(L)])
QB = (v.T * E * v)[0]
def fhat(r):
    s = 0
    for coef, om in ((w[1], w[0]), (w[0], w[1])):
        s += coef * (mp.sin((om - r) * L) / (om - r) + mp.sin((om + r) * L) / (om + r))
    return s
zs = rhlib.zeros(5000)
QZ = 2 * sum(fhat(mp.mpf(g)) ** 2 for g in zs)
print("  B (prime side): %s   zero side (5000 zeros): %s   rel diff %s" % (mp.nstr(QB, 15), mp.nstr(QZ, 15), mp.nstr(abs(QB - QZ) / abs(QZ), 3)))
E2, _ = full(L, 2, 'half', bl)
QC = (v.T * E2 * v)[0]
print("  control Beurling 2->2.1: %s  rel diff to zero side %s" % (mp.nstr(QC, 15), mp.nstr(abs(QC - QZ) / abs(QZ), 3)))
print("  odd test: f = sin(w0 x): B %s" % mp.nstr(O[0, 0] * L, 15), end="")
def fhs(r):  # int sin(w0 x) sin(r x) dx
    om = w[0]; return mp.sin((om - r) * L) / (om - r) - mp.sin((om + r) * L) / (om + r)
print("   zero side %s (slow tail, expect ~1e-3 agreement)" % mp.nstr(2 * sum(fhs(mp.mpf(g)) ** 2 for g in zs), 15))
print("total %.1fs" % (time.time() - t0))
