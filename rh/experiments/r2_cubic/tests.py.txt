import sys, math, importlib.util, importlib.machinery, numpy as np, mpmath as mp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m

np.set_printoptions(linewidth=200, precision=4, suppress=True)
def sig(M, tol=1e-12):
    e = np.linalg.eigvalsh(M); return int((e > tol).sum()), e[::-1]
def cone(H, a):
    """cone certificate: T_ijk = [ H_ij l_k + H_ik l_j + H_jk l_i - (2/s) l_i l_j l_k ] / s ,  l = H a, s = a.l ;  T.a = H"""
    l = H @ a; s = a @ l
    T = (np.einsum('ij,k->ijk', H, l) + np.einsum('ik,j->ijk', H, l) + np.einsum('jk,i->ijk', H, l) - (2 / s) * np.einsum('i,j,k->ijk', l, l, l)) / s
    return T
which = sys.argv[1]
if which == "assembly":
    L, n = 0.8, 8
    Nb, p, q = box.boxN(L, n); h = 2 * L / n
    rng = np.random.default_rng(1); x = rng.normal(size=n)
    step = lambda t: mp.mpf(x[min(n - 1, max(0, int(mp.floor((t + L) / h))))] / math.sqrt(h)) if -L <= t <= L else mp.mpf(0)
    # independent route: correlation C(u) of the step function by direct quadrature with breakpoints, then the formulas
    cn = box.lam(math.exp(2 * L))
    def C(u):
        br = sorted(set([-L + i * h for i in range(n + 1)] + [-L + i * h - float(u) for i in range(n + 1)]))
        br = [b for b in br if -L <= b <= L - float(u)]
        if len(br) < 2: return mp.mpf(0)
        return mp.quad(lambda t: step(t + u + mp.mpf(10) ** -25) * step(t + mp.mpf(10) ** -25), br)
    g0 = C(0)
    grid = [i * h / 2 for i in range(0, 4 * n + 1)]
    archv = -mp.log(mp.pi) * g0 + mp.quad(lambda u: g0 * mp.exp(-2 * u) / u - 2 * box.w(u) * C(u), grid) + g0 * mp.e1(4 * L)
    pr = sum(c / mp.sqrt(xx) * 2 * C(mp.log(xx)) for xx, c in cn.items())
    print("assembly check, random step function, L=0.8 n=8:  matrix %.12f   direct quadrature %.12f" % (x @ Nb @ x, float(pr - archv)))

    for L, n in ((0.5, 12), (1.0, 12)):
        Nb, p, q = box.boxN(L, n); H = box.ext(Nb, p, q)
        ev, V = np.linalg.eigh(H); a = np.abs(V[:, -1])
        T = cone(H, a)
        print("L=%.2f n=%d  Perron a: boxes %s fibres %s ; (Ha)min %.3e ; |T.a - H| %.1e" % (L, n, a[:3], a[-2:], (H @ a).min(), np.abs(np.einsum('ijk,k->ij', T, a) - H).max()))

        idx = [(i, j, k) for i in range(n) for j in range(i + 1, n) for k in range(j + 1, n)]
        vals = np.array([T[i, j, k] for i, j, k in idx])
        print("   distinct box triples: %d, negative: %d (%.0f%%)" % (len(vals), (vals < 0).sum(), 100 * (vals < 0).mean()))
