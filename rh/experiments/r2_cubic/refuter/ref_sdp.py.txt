import sys, math, itertools, importlib.machinery, importlib.util, numpy as np, cvxpy as cp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm
R = load("ref_lib", "/home/user/ysolucki/rh/experiments/r2_cubic/refuter/ref_lib.py.txt")
def Hmat(kind, L, n):
    at = None if kind == "zeta" else R.beurling(math.exp(2 * L))
    N, p, q, _ = R.build(L, n, at); return R.ext(N, p, q)

def cert(H, a, mode, zero=(), solver="CLARABEL"):
    """own formulation: PSD variables P_k; T_k = l_k l_k^T/s_k - P_k; constraints: full symmetry of T, sum a_k T_k = H.
    mode 'sign': maximise t with T_ijk >= t l_i l_j l_k / s^2 on distinct triples, P_k >= 0.
    mode 'slack': maximise t with P_k - t I >= 0 (t <= 1), entries in `zero` forced to 0."""
    m = len(a); l = H @ a; s = float(a @ l)
    Pv = [cp.Variable((m, m), symmetric=True) for _ in range(m)]
    T = [np.outer(H[:, k], H[:, k]) / l[k] - Pv[k] for k in range(m)]
    t = cp.Variable(); cons = []
    for k in range(m):
        for i in range(k):
            cons.append(T[k][i, :] == T[i][k, :])        # T_ijk symmetric under swapping a slice index with a matrix index
    cons.append(sum(a[k] * T[k] for k in range(m)) == H)
    if mode == "sign":
        for k in range(m): cons.append(Pv[k] >> 0)
        for i, j, k in itertools.combinations(range(m), 3):
            cons.append(T[k][i, j] >= t * l[i] * l[j] * l[k] / s ** 2)
        cons.append(t <= 10)
    else:
        for k in range(m): cons.append(Pv[k] - t * np.eye(m) >> 0)
        for (i, j, k) in zero: cons.append(T[k][i, j] == 0)
        cons.append(t <= 1)
    prob = cp.Problem(cp.Maximize(t), cons)
    try: prob.solve(solver=solver)
    except Exception as ex: return "fail:" + type(ex).__name__, None, None
    if t.value is None: return prob.status, None, None
    Tn = np.zeros((m, m, m))
    for k in range(m): Tn[:, :, k] = np.outer(H[:, k], H[:, k]) / l[k] - Pv[k].value
    return prob.status, float(t.value), Tn

def check(H, a, T):
    m = len(a); l = H @ a
    sym = max(np.abs(T - np.transpose(T, pp)).max() for pp in itertools.permutations(range(3)))
    res = np.abs(np.einsum('ijk,k->ij', T, a) - H).max()
    npos = max(int((np.linalg.eigvalsh(T[:, :, k]) > 1e-7).sum()) for k in range(m))
    return "sym %.0e |T.a-H| %.0e max#pos %d" % (sym, res, npos)

            print("%s L=%.2f n=%d a=Perron %s: %s t*=%s %s" % (kind, L, n, solver, st, "%.5f" % t if t is not None else "-", check(H, a, T) if T is not None else ""), flush=True)
