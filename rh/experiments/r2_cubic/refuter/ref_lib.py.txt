"""R2-cubic-ref: independent box builder (scipy quad, no mpmath, no import of the seat's code).
N = primes - arch in the orthonormal box basis on [-L,L]; p,q; Ntilde."""
import math, numpy as np
from scipy.integrate import quad
from scipy.special import exp1

def w(u):  # e^{u/2}/(2 sinh u) written stably
    return math.exp(-u / 2) / (-math.expm1(-2 * u))

def vonmangoldt(X):
    out = []
    for p in range(2, int(X) + 1):
        if all(p % d for d in range(2, int(p ** 0.5) + 1)):
            pk = p
            while pk <= X + 1e-12:
                out.append((float(pk), math.log(p))); pk *= p
    return out

def beurling(X, g=2.1):
    out = []
    for p in [g] + [r for r in range(3, int(X) + 2) if all(r % d for d in range(2, int(r ** 0.5) + 1))]:
        pk = p
        while pk <= X + 1e-12:
            out.append((float(pk), math.log(p))); pk *= p
    return out

def build(L, n, atoms=None):
    h = 2.0 * L / n
    if atoms is None: atoms = vonmangoldt(math.exp(2 * L))
    tri = lambda v: max(0.0, 1.0 - abs(v) / h)
    arch = np.zeros(n)
    # diagonal: -log pi + int_0^h (e^{-2u}/u - 2w(u)(1-u/h)) du + E1(2h)
    f0 = lambda u: (math.exp(-2 * u) / u - 2 * w(u) * (1 - u / h)) if u > 0 else 0.0
    arch[0] = -math.log(math.pi) + quad(f0, 0, h, epsabs=1e-13, epsrel=1e-13, limit=200)[0] + exp1(2 * h)
    for m in range(1, n):
        d = m * h
        lo = d - h
        v1 = quad(lambda u: w(u) * tri(u - d), max(lo, 0.0), d, epsabs=1e-13, epsrel=1e-13, limit=200)[0]
        v2 = quad(lambda u: w(u) * tri(u - d), d, d + h, epsabs=1e-13, epsrel=1e-13, limit=200)[0]
        arch[m] = -(v1 + v2)
    N = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            d = abs(i - j) * h
            pr = 0.0
            for x, c in atoms:
                u = math.log(x)
                pr += c / math.sqrt(x) * (tri(u - d) + tri(-u - d))
            N[i, j] = pr - arch[abs(i - j)]
    xs = -L + h * np.arange(n + 1)
    p = 2 * (np.exp(xs[1:] / 2) - np.exp(xs[:-1] / 2)) / math.sqrt(h)
    q = 2 * (np.exp(-xs[:-1] / 2) - np.exp(-xs[1:] / 2)) / math.sqrt(h)
    return N, p, q, arch

def ext(N, p, q):
    n = len(p); M = np.zeros((n + 2, n + 2))
    M[:n, :n] = N; M[:n, n] = M[n, :n] = p; M[:n, n + 1] = M[n + 1, :n] = q; M[n, n + 1] = M[n + 1, n] = 1
    return M

def perron(H):
    ev, V = np.linalg.eigh(H); a = V[:, -1]; a = a * np.sign(a.sum())
    return ev[-1], a
