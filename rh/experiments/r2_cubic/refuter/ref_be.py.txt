def gammas(H, a):
    """site forms G1[k], G2[k] with Gamma(f)(k) = f^T G1[k] f, Gamma_2(f)(k) = f^T G2[k] f; built from L = K - I by definition
    (via the bilinear carre du champ 2 Gamma(f,g) = L(fg) - f Lg - g Lf written as tensors)."""
    m = len(a); s = H @ a
    K = (H * a[None, :]) / s[:, None]; Lg = K - np.eye(m)
    # C[k,i,j]: Gamma(f,g)(k) = sum_ij C[k,i,j] f_i g_j ;  L(fg)(k) = sum_i L_ki f_i g_i
    C = np.zeros((m, m, m))
    for k in range(m):
        C[k] += 0.5 * np.diag(Lg[k])
        C[k, k, :] -= 0.5 * Lg[k]
        C[k, :, k] -= 0.5 * Lg[k]
    G1 = C.copy()
    # Gamma_2(f)(k) = 1/2 [ sum_i L_ki Gamma(f)(i) - 2 Gamma(f, Lf)(k) ]
    G2 = np.zeros((m, m, m))
    LG = np.einsum('ki,iab->kab', Lg, G1)
    for k in range(m):
        X = G1[k] @ Lg            # f^T G1[k] (L f)
        G2[k] = 0.5 * (LG[k] - (X + X.T))
    return K, Lg, G1, G2

def rho_be(G1, G2):
    m = G1.shape[0]
    # basis of mean-zero-free quotient: drop constants via an orthonormal complement of 1
    Qc = sla.null_space(np.ones((1, m)))
    best = np.inf
    for k in range(m):
        A = Qc.T @ G2[k] @ Qc; B = Qc.T @ G1[k] @ Qc
        ev = sla.eigh(A, B, eigvals_only=True)
        best = min(best, ev[0])
    return best

def tbe(H, a):
    m = len(a); s = H @ a; pi = a * s
    K, Lg, G1, G2 = gammas(H, a)
    A1 = np.diag(1 / a)
    T = np.zeros((m, m, m))
    for k in range(m):
        T[:, :, k] = np.outer(H[:, k], H[:, k]) / s[k] - s[k] * (A1 @ (G2[k] - G1[k]) @ A1)
    return T, K, G1, G2

def theta_def(H, a):
    """Theta of Theorem F from its DEFINITION (trilinear products), as a tensor in x-coordinates."""
    m = len(a); s = H @ a; pi = a * s
    K = (H * a[None, :]) / s[:, None]
    U0 = np.diag(1 / a); U1 = K @ U0; U2 = K @ U1
    br = lambda X, Y, Z: np.einsum('l,li,lj,lk->ijk', pi, X, Y, Z)
    return 0.5 * (br(U0, U1, U1) + br(U1, U0, U1) + br(U1, U1, U0)) - 0.25 * (br(U2, U0, U0) + br(U0, U2, U0) + br(U0, U0, U2)) + 0.25 * br(U0, U0, U0)

def f5(H, a):
    m = len(a); s = H @ a; HDH = H @ np.diag(a / s) @ H
    T = np.zeros((m, m, m))
    for i, j, k in itertools.product(range(m), repeat=3):
        if len({i, j, k}) == 3:
            T[i, j, k] = 0.5 * (H[i, j] * H[i, k] / s[i] + H[i, j] * H[j, k] / s[j] + H[i, k] * H[j, k] / s[k])
        elif i == j == k:
            T[i, j, k] = 1.5 * H[i, i] ** 2 / s[i] - 0.75 * HDH[i, i] / a[i] + s[i] / (4 * a[i] ** 2)
        else:
            u = i if (i == j or i == k) else j
            v = [t for t in (i, j, k) if t != u][0]
            T[i, j, k] = H[u, u] * H[u, v] / s[u] + H[u, v] ** 2 / (2 * s[v]) - HDH[u, v] / (4 * a[u])
    return T

def symdef(T):
    return max(np.abs(T - np.transpose(T, pp)).max() for pp in itertools.permutations(range(3)))
