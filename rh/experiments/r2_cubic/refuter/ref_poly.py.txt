def Hmat(kind, L, n):
    at = None if kind == "zeta" else R.beurling(math.exp(2 * L))

def family(H, a, d):
    """symmetrised [K^a g, K^b f, K^c h]_pi, 0<=a<=b<=c<=d, in x-coordinates; then an ORTHONORMAL basis of their span (QR, pivoted)."""
    m = len(a); s = H @ a; pi = a * s; K = (H * a[None, :]) / s[:, None]
    U = [np.diag(1 / a)]
    for _ in range(d): U.append(K @ U[-1])
    raw = []
    for (i, j, k) in itertools.combinations_with_replacement(range(d + 1), 3):
        t = np.einsum('l,lx,ly,lz->xyz', pi, U[i], U[j], U[k])
        raw.append(sum(np.transpose(t, pp) for pp in itertools.permutations(range(3))) / 6)
    M = np.array([r.reshape(-1) / np.linalg.norm(r) for r in raw]).T
    Q, Rr, piv = sla.qr(M, mode='economic', pivoting=True)
    rank = int((np.abs(np.diag(Rr)) > 1e-11 * abs(Rr[0, 0])).sum())
    return [Q[:, i].reshape(m, m, m) for i in range(rank)], len(raw), np.abs(np.diag(Rr))
def solve(H, a, d, mode, solver="CLARABEL"):
    m = len(a); s = H @ a; fam, nraw, rd = family(H, a, d)
    c = cp.Variable(len(fam)); t = cp.Variable()
    ah = a / np.linalg.norm(a); Pr = np.eye(m) - np.outer(ah, ah)
    cons = [sum(c[i] * np.einsum('ijk,k->ij', F, a) for i, F in enumerate(fam)) == H]
    for k in range(m):
        Tk = sum(c[i] * F[:, :, k] for i, F in enumerate(fam))
        Pk = np.outer(H[:, k], H[:, k]) / s[k] - Tk
        cons.append(Pk - t * (Pr if mode == "margin" else np.eye(m)) >> 0)
    cons.append(t <= 1)
    prob = cp.Problem(cp.Maximize(t), cons)
    try: prob.solve(solver=solver)
    except Exception as ex: return "fail", None, None, len(fam)
    if c.value is None: return prob.status, None, None, len(fam)
    T = sum(c.value[i] * F for i, F in enumerate(fam))
    # independent check of the returned tensor: eigenvalues of P_k on the complement of a
    B = sla.null_space(a[None, :])
    lam_perp = min(np.linalg.eigvalsh(B.T @ (np.outer(H[:, k], H[:, k]) / s[k] - T[:, :, k]) @ B)[0] for k in range(m))
    lam_full = min(np.linalg.eigvalsh(np.outer(H[:, k], H[:, k]) / s[k] - T[:, :, k])[0] for k in range(m))
    npos = max(int((np.linalg.eigvalsh(T[:, :, k]) > 1e-9).sum()) for k in range(m))
    res = np.abs(np.einsum('ijk,k->ij', T, a) - H).max()
    return prob.status, float(t.value), (lam_perp, lam_full, npos, res), len(fam)
