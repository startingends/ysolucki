import sys, math, itertools, importlib.util, importlib.machinery, numpy as np, cvxpy as cp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm
p2 = load("p2", "/home/user/ysolucki/rh/experiments/r2_cubic/poly2lib.py.txt")
rng = np.random.default_rng(7)
for m in (8, 12):
    for trial in range(6):
        v = rng.uniform(0.5, 1.5, m); G = rng.normal(size=(m, m)) * 0.25; B = G @ G.T
        B = B - np.diag(np.diag(B)); B = np.minimum(B, 0.9 * np.outer(v, v))     # keep off-diagonal of v v^T - B >= 0
        H = np.outer(v, v) - B - np.diag(rng.uniform(0.5, 3.0, m))
        ev, V = np.linalg.eigh(H)
        # shift diagonal down until exactly one positive eigenvalue
        while (ev > 1e-10).sum() > 1:
            H -= 0.2 * np.eye(m); ev, V = np.linalg.eigh(H)
        a = np.abs(V[:, -1])
        if ev[-1] <= 0 or (H @ a).min() <= 0: print("skip"); continue
        out = []
        for d in (1, 2):
            st, tau, chk, co = p2.solve(H, a, d, cp.CLARABEL)
            if chk is None: st, tau, chk, co = p2.solve(H, a, d, cp.SCS)
            out.append("d=%d %s tau*=%s lam_min(P)=%s #pos=%s" % (d, st[:9], "%.2e" % tau if tau is not None else "nan", "%.2e" % chk[0] if chk else "-", chk[1] if chk else "-"))
        print("m=%d trial %d eig2(H)=%.3f | " % (m, trial, ev[-2]) + " | ".join(out), flush=True)
