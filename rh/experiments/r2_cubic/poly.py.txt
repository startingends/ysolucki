import sys, math, itertools, importlib.util, importlib.machinery, numpy as np, cvxpy as cp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm
box = load("box", "/home/user/ysolucki/rh/experiments/r2_cubic/box.py.txt")
def family(H, a, d):
    m = len(a); s = H @ a; pi = a * s
    U = [np.diag(1 / a)]; Kx = (H * a[None, :]) / s[:, None]          # K in f-coordinates
    for _ in range(d): U.append(Kx @ U[-1])
    out = {}
    for (c, b, aa) in itertools.combinations_with_replacement(range(d + 1), 3):
        t = np.einsum('l,li,lj,lk->ijk', pi, U[aa], U[b], U[c])
        out[(aa, b, c)] = sum(np.transpose(t, pp) for pp in itertools.permutations(range(3))) / 6
    return out
def solve(H, a, d):
    m = len(a); s = H @ a; fam = family(H, a, d); keys = list(fam)
    c = cp.Variable(len(keys)); tau = cp.Variable()
    cons = [sum(c[i] * np.einsum('ijk,k->ij', fam[k], a) for i, k in enumerate(keys)) == H, tau >= 0]
    for k in range(m):
        Tk = sum(c[i] * fam[key][:, :, k] for i, key in enumerate(keys))
        cons.append(np.outer(H[:, k], H[:, k]) / s[k] - Tk + tau * np.eye(m) >> 0)
    prob = cp.Problem(cp.Minimize(tau + 1e-9 * cp.sum_squares(c)), cons)
    try: prob.solve(solver=cp.CLARABEL)
    except Exception as ex: return "fail", None, None
    return prob.status, tau.value, dict(zip(keys, c.value)) if c.value is not None else None
for kind, L, n in (("zeta", 1.0, 12), ("zeta", 0.5, 12), ("beurling", 1.0, 12), ("beurling", 0.5, 8), ("zeta", 1.0, 20)):
    cn = box.lam(math.exp(2 * L)) if kind == "zeta" else box.beurling(math.exp(2 * L))
    Nb, p, q = box.boxN(L, n, cn); H = box.ext(Nb, p, q); ev, V = np.linalg.eigh(H); a = np.abs(V[:, -1])
    line = "%s L=%.2f n=%d #pos(H)=%d : " % (kind, L, n, (ev > 1e-10).sum())
    for d in (2, 3, 4):
        st, tau, co = solve(H, a, d)
        line += " d=%d tau*=%s (%s)" % (d, "%.3e" % tau if tau is not None else "nan", st[:9])
        if kind == "zeta" and L == 1.0 and n == 12 and d == 2 and co: print("   d=2 coefficients:", {k: round(float(v), 4) for k, v in co.items()})
    print(line, flush=True)
