"""Stage 2: the nine promotions. Fields are filled in the LOCAL entry first, then promoted, then the judge's status is set.
Cross-references use @name@ placeholders resolved from the main numbers actually returned by the tool (backward only)."""
import json, os, sys
exec(open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "audit_lib.py.txt")).read())
marker = once("stage2")
M = {}


def R(s):
    for k, v in M.items():
        s = s.replace("@%s@" % k, "#%d" % v)
    assert "@" not in s, s
    return s


def P(name, local, n, fields, status, why):
    out_fields = {f: R(t) for f, t in fields.items()}
    replaced = fill(local, n, out_fields)
    out = run("promote", "--from", local, str(n), "--auditor", AUD)
    m = int(out.rsplit("#", 1)[1])
    M[name] = m
    run("status", str(m), status, R(why))
    for f, old in replaced.items():
        run("--file", local, "note", str(n), "auditor %s replaced the '%s' field before promotion; the agent's original text was: %s" % (AUD, f, old))
    json.dump(M, open(os.path.join(HERE, "main_numbers.json"), "w"), indent=1)
