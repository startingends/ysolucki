"""Baez-Duarte / Nyman-Beurling Gram matrix via Vasyunin's cotangent-sum formula.
e_k(x) = frac(1/(k x)) in L^2(0,inf).  G[h,k] = <e_h,e_k> = int_0^inf frac(t/h) frac(t/k) dt/t^2.
For gcd(h,k)=1 (Vasyunin 1995, written from memory, VALIDATED below by direct integration):
  G[h,k] = (log(2pi)-gamma)/2 * (1/h+1/k) + (k-h)/(2hk) * log(h/k) - pi/(2hk) * (V(h,k)+V(k,h)),
  V(h,k) = sum_{m=1}^{k-1} frac(m h/k) cot(pi m/k).
For g=gcd(h,k)>1: G[h,k] = G[h/g,k/g]/g (scaling).   b_k = <chi_(0,1), e_k> = (1-gamma+log k)/k.
"""
import sys, math, time
import numpy as np
EG = 0.57721566490153286061

def vtable(N):
    """vt[k][r] = V(r,k) for r in 0..k-1 (depends on h mod k only)."""
    vt = [None, np.zeros(1)]
    for k in range(2, N + 1):
        m = np.arange(1, k, dtype=np.int64)
        cot = 1.0 / np.tan(np.pi * m / k)
        r = np.arange(0, k, dtype=np.int64)
        fr = (np.outer(r, m) % k) / k          # shape (k, k-1)
        vt.append(fr @ cot)
    return vt

def gram(N, vt=None):
    if vt is None:
        vt = vtable(N)
    c0 = (math.log(2 * math.pi) - EG) / 2
    G = np.zeros((N + 1, N + 1))
    idx = np.arange(1, N + 1)
    for k in range(1, N + 1):
        h = np.arange(1, k + 1)
        # V(h,k): index h mod k ; V(k,h): vt[h][k mod h]
        Vhk = vt[k][h % k]
        Vkh = np.array([vt[int(hh)][k % int(hh)] for hh in h])
        val = c0 * (1.0 / h + 1.0 / k) + (k - h) / (2.0 * h * k) * np.log(h / k) - np.pi / (2.0 * h * k) * (Vhk + Vkh)
        G[1:k + 1, k] = val
    # fix non-coprime entries by scaling, then symmetrise
    for k in range(1, N + 1):
        for h in range(1, k + 1):
            g = math.gcd(h, k)
            if g > 1:
                G[h, k] = G[h // g, k // g] / g
    G = G[1:, 1:]
    G = np.triu(G) + np.triu(G, 1).T
    return G

def bvec(N):
    k = np.arange(1, N + 1)
    return (1 - EG + np.log(k)) / k

def direct_entry(h, k, T=200000):
    """Independent check: exact piecewise integration of frac(t/h)frac(t/k)/t^2 on [0,T*h*k] + mean-value tail."""
    import mpmath as mp
    mp.mp.dps = 30
    L = h * k // math.gcd(h, k)
    pts = sorted(set(list(range(0, T * L + 1, h)) + list(range(0, T * L + 1, k))))
    tot = mp.mpf(0)
    # vectorised in float128-ish via numpy longdouble
    p = np.array(pts, dtype=np.longdouble)
    lo, hi = p[:-1], p[1:]
    a = np.floor(np.array(pts[:-1]) / h).astype(np.longdouble)
    b = np.floor(np.array(pts[:-1]) / k).astype(np.longdouble)
    hh, kk = np.longdouble(h), np.longdouble(k)
    # integrand (t/h-a)(t/k-b)/t^2 = 1/(hk) - (a/k+b/h)/t + ab/t^2
    seg = np.zeros_like(lo)
    seg += (hi - lo) / (hh * kk)
    mask = lo > 0
    seg[mask] += -(a[mask] / kk + b[mask] / hh) * np.log(hi[mask] / lo[mask]) + a[mask] * b[mask] * (1 / lo[mask] - 1 / hi[mask])
    s = float(np.sum(seg))
    # tail: mean of frac(t/h)frac(t/k) over a period = 1/4 + g^2/(12 h k)
    g = math.gcd(h, k)
    mean = 0.25 + g * g / (12.0 * h * k)
    return s + mean / (T * L)

if __name__ == "__main__":
    mode = sys.argv[1]
    if mode == "validate":
        G = gram(12)
        worst = 0
        for h in range(1, 13):
            for k in range(h, 13):
                d = direct_entry(h, k, T=20000)
                worst = max(worst, abs(d - G[h - 1, k - 1]))
                if (h, k) in [(1, 1), (1, 2), (2, 3), (3, 7), (4, 6), (5, 12), (11, 12)]:
                    print(h, k, "vasyunin %.12f direct %.12f diff %.2e" % (G[h - 1, k - 1], d, d - G[h - 1, k - 1]))
        print("worst |diff| over h<=k<=12:", worst)
    elif mode == "build":
        N = int(sys.argv[2]); out = sys.argv[3]
        t0 = time.time(); vt = vtable(N); print("vtable", time.time() - t0, flush=True)
        G = gram(N, vt); print("gram", time.time() - t0, flush=True)
        np.save(out, G)
