"""de Bruijn-Newman: zeros of H_t(z) = int_0^inf e^{t u^2} Phi(u) cos(z u) du at low height; ODE check.
Phi(u) = sum_{n>=1} (2 pi^2 n^4 e^{9u} - 3 pi n^2 e^{5u}) exp(-pi n^2 e^{4u}).  H_0(z) = xi(1/2 + i z/2)/8, zeros z = 2*gamma.
"""
import sys, json
import mpmath as mp
mp.mp.dps = 90

def Phi(u):
    s = mp.mpf(0)
    e4 = mp.exp(4 * u)
    for n in range(1, 30):
        term = (2 * mp.pi**2 * n**4 * mp.exp(9 * u) - 3 * mp.pi * n**2 * mp.exp(5 * u)) * mp.exp(-mp.pi * n * n * e4)
        s += term
        if abs(term) < mp.mpf(10)**(-60) * abs(s):
            break
    return s

# fixed Gauss-Legendre grid on [0, U]; Phi decays doubly exponentially, U=1.6 is ample (Phi(1.6) ~ exp(-pi e^6.4) ~ 1e-820)
U = mp.mpf("1.3")
NODES = None
def setup(npts=600):
    global NODES
    xs, ws = [], []
    # composite Gauss-Legendre: 30 panels x 20 nodes
    panels = 30
    gl = mp.calculus.quadrature.GaussLegendre(mp.mp)
    nodes = gl.calc_nodes(5, mp.mp.prec)  # degree 5 -> 3*2^4=48 nodes on [-1,1]
    for p in range(panels):
        a = U * p / panels; b = U * (p + 1) / panels
        for x, w in nodes:
            xs.append((a + b) / 2 + (b - a) / 2 * x); ws.append(w * (b - a) / 2)
    NODES = [(x, w * Phi(x)) for x, w in zip(xs, ws)]

def H(t, z):
    return mp.fsum(wp * mp.exp(t * x * x) * mp.cos(z * x) for x, wp in NODES)

def zeros_Ht(t, guesses):
    out = []
    for g in guesses:
        out.append(mp.findroot(lambda z: H(t, z), g))
    return out

if __name__ == "__main__":
    sys.path.insert(0, "/home/user/ysolucki/tools"); import rhlib
    K = int(sys.argv[1]) if len(sys.argv) > 1 else 40
    setup()
    gam = rhlib.zeros(K)
    # sanity: H_0(z) vs xi(1/2+iz/2)/8
    for z in [mp.mpf(3), mp.mpf(20), mp.mpf(60)]:
        print("H0 check z=%s: %s vs %s" % (z, mp.nstr(H(0, z), 20), mp.nstr(rhlib.xi(mp.mpf(1) / 2 + 1j * z / 2).real / 8, 20)))
    ts = [mp.mpf(i) / 20 for i in range(0, 11)]   # 0, 0.05, ..., 0.5
    res = {}
    guess = [2 * mp.mpf(g) for g in gam]
    for t in ts:
        zs = zeros_Ht(t, guess)
        guess = zs
        res[str(float(t))] = [mp.nstr(z, 25) for z in zs]
        print("t=%.2f first zeros: %s  min gap %.6f" % (float(t), [mp.nstr(z, 12) for z in zs[:4]], float(min(zs[i + 1] - zs[i] for i in range(len(zs) - 1)))), flush=True)
    json.dump(res, open("/home/user/ysolucki/rh/data/r1_B/dbn_zeros.json", "w"))
