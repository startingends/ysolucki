"""Critic: independent Gram matrix for e_k(y)=frac(y/k) in L^2((0,inf), dy/y^2), no Vasyunin.
Exact piecewise integration on [0,Y) + mean-value tail (1/4 + g^2/(12hk))/Y."""
import sys, math, numpy as np, time
EG = 0.57721566490153286061
N = int(sys.argv[1]); Y = int(float(sys.argv[2]))
k = np.arange(1, N + 1)
S1 = np.zeros(N); S2 = np.zeros((N, N)); SB = np.zeros(N)
t0 = time.time(); CH = 50000
for a in range(1, Y, CH):
    n = np.arange(a, min(a + CH, Y))
    A = (n[:, None] // k[None, :]).astype(np.float64)
    L = np.log1p(1.0 / n); D = 1.0 / (n * (n + 1.0))
    S1 += A.T @ L; S2 += A.T @ (A * D[:, None]); SB += A.T @ D
h = k[:, None]; kk = k[None, :]
g = np.gcd(h, kk)
G = Y / (h * kk) - S1[:, None] / kk - S1[None, :] / h + S2 + (0.25 + g * g / (12.0 * h * kk)) / Y
b = math.log(Y) / k - SB + 0.5 / Y
print("built N=%d Y=%d in %.0fs" % (N, Y, time.time() - t0))
print("G[1,1] = %.10f  (log 2pi - gamma = %.10f)" % (G[0, 0], math.log(2 * math.pi) - EG))
print("max |b - (1-gamma+log k)/k| = %.2e" % np.max(np.abs(b - (1 - EG + np.log(k)) / k)))
Gp = np.load("/home/user/ysolucki/rh/data/r1_B/G2000.npy")[:N, :N]
print("max |G_critic - G_proposer| on %dx%d block: %.3e" % (N, N, np.max(np.abs(G - Gp))))
bb = (1 - EG + np.log(k)) / k
for M in [10, 50, 100, 200]:
    if M > N: continue
    c = np.linalg.solve(G[:M, :M], bb[:M]); d2 = 1 - bb[:M] @ c
    cp = np.linalg.solve(Gp[:M, :M], bb[:M]); d2p = 1 - bb[:M] @ cp
    # V_N recipe
    from sympy import mobius
    mu = np.array([int(mobius(int(j))) for j in k[:M]])
    v = -mu * (1 - np.log(k[:M]) / math.log(M))
    Q = 1 - 2 * bb[:M] @ v + v @ G[:M, :M] @ v
    print("N=%4d d_N^2 critic %.10f  proposerG %.10f  d^2 logN %.4f | Q[V_N] critic %.7f" % (M, d2, d2p, d2 * math.log(M), Q))
np.save("/tmp/claude-0/-home-user-ysolucki/9039aa06-6540-523a-ba98-9efe520848c8/scratchpad/Gcritic.npy", G)
