G = np.load("/home/user/ysolucki/rh/data/r1_B/G2000.npy"); Nmax = G.shape[0]
b = bvec(Nmax); mu = np.array([int(mobius(k)) for k in range(1, Nmax + 1)], dtype=float)
ks = np.arange(1, Nmax + 1); lk = np.log(ks)
def Q(c, N): return 1 - 2 * b[:N] @ c + c @ G[:N, :N] @ c
def Qsub(B, N):
    """min of Q over span of columns of B (N x m). returns (Qmin, coeffs)."""
    A = B.T @ G[:N, :N] @ B; r = B.T @ b[:N]; a = np.linalg.solve(A, r); return 1 - r @ a, a
# NOTE sign: sum_k mu(k) frac(1/(kx)) = -chi formally, so natural ansatz is c_k = -mu(k) * cutoff.
print("SIGN-CORRECTED. columns: N | d_N^2 | Q[V_N], V_N=-mu(1-logk/logN) | Q*logN | best M-family Q, logM/logN | best eps-family Q[-mu k^-eps], eps*logN")
for N in [50, 100, 200, 500, 1000, 2000]:
    copt = np.linalg.solve(G[:N, :N], b[:N]); dopt = 1 - b[:N] @ copt
    v = -mu[:N] * (1 - lk[:N] / math.log(N)); qv = Q(v, N)
    best = (1e9, None)
    for th in np.linspace(0.8, 3, 441):
        w = -mu[:N] * (1 - lk[:N] / (th * math.log(N))); q = Q(w, N)
        if q < best[0]: best = (q, th)
    beste = (1e9, None)
    for e in np.linspace(0.0, 3, 601):
        w = -mu[:N] * ks[:N] ** (-e / math.log(N)); q = Q(w, N)
        if q < beste[0]: beste = (q, e)
    print("%5d | %.7f | %.7f | %.5f | %.7f th=%.3f | %.7f e=%.3f" % (N, dopt, qv, qv * math.log(N), best[0], best[1], beste[0], beste[1]))
