"""K-atlas: sign changes of folded lowest eigenvectors (zeta, DH), and zeta with a prime removed.
usage: python3 signs.py.txt gs <kind> <L> <N> <bits>      kind = zeta | dh
       python3 signs.py.txt rm <q> <N> <dps> L1 L2 ...     zeta with prime q removed (and control with q present)"""
import sys, math, importlib.util, importlib.machinery
import flint
from flint import arb
P = "/home/user/ysolucki/rh/experiments/r1_A/weil.py.txt"
spec = importlib.util.spec_from_file_location("weil", P, loader=importlib.machinery.SourceFileLoader("weil", P))
weil = importlib.util.module_from_spec(spec); spec.loader.exec_module(weil)

def fold_signs(vec, L, parity, npts=2000):
    c = [float(v.mid()) for v in vec]
    vals = []
    for i in range(1, npts):
        x = L * i / npts
        if parity == 'even':
            f = c[0] + sum(c[k] * math.sqrt(2) * math.cos(math.pi * k * x / L) for k in range(1, len(c)))
        else:
            f = sum(c[k - 1] * math.sqrt(2) * math.sin(math.pi * k * x / L) for k in range(1, len(c) + 1))
        vals.append(f)
    m = max(abs(v) for v in vals)
    sg = [1 if v > 0 else -1 for v in vals if abs(v) > 1e-9 * m]
    S = sum(1 for a, b in zip(sg, sg[1:]) if a != b)
    k = max(range(len(c)), key=lambda j: abs(c[j]))
    tot = sum(v * v for v in c)
    return S, min(vals) / m, max(vals) / m, k, c[k] ** 2 / tot

if sys.argv[1] == 'gs':
    kind, L, N, bits = sys.argv[2], float(sys.argv[3]), int(sys.argv[4]), int(sys.argv[5])
    flint.ctx.prec = bits
    q, w = weil.build(L, N, kind=kind)
    for name, blk in (('even', weil.even_block(q, N)), ('odd', weil.odd_block(q, N))):
        mu, vec, _ = weil.lowest(blk)
        S, lo, hi, k, share = fold_signs(vec, L, name)
        kk = k if name == 'even' else k + 1
        print("%s L=%g N=%d %s: eig of smallest modulus %s ; sign changes on (0,L) = %d ; min/max of f over max|f| = %.3g / %.3g ; largest mode k=%d (freq %.2f) share %.3f"
              % (kind, L, N, name, mu.str(6, radius=False), S, lo, hi, kk, math.pi * kk / L, share), flush=True)
else:
    import mpmath as mp
    qrem, N, dps = int(sys.argv[2]), int(sys.argv[3]), int(sys.argv[4])
    mp.mp.dps = dps; flint.ctx.prec = int(dps * 3.33) + 40
    orig = weil.mangoldt_upto
    def to_mp(M):
        n = M.nrows(); R = mp.matrix(n, n)
        for i in range(n):
            for j in range(n):
                R[i, j] = mp.mpf(M[i, j].mid().str(dps + 10, radius=False))
        return R
    for Ls in sys.argv[5:]:
        L = float(Ls)
        for label, filt in (("without %d" % qrem, True), ("control (all primes)", False)):
            if filt:
                weil.mangoldt_upto = lambda X: {n: v for n, v in orig(X).items() if v[1] != qrem}
            else:
                weil.mangoldt_upto = orig
            q, w = weil.build(L, N, kind='zeta')
            ev = min(mp.eigsy(to_mp(weil.even_block(q, N)), eigvals_only=True))
            od = min(mp.eigsy(to_mp(weil.odd_block(q, N)), eigvals_only=True))
            print("L=%.4f %s: lowest eigenvalue even %s  odd %s" % (L, label, mp.nstr(ev, 5), mp.nstr(od, 5)), flush=True)
        weil.mangoldt_upto = orig
