"""K-atlas: Q_bg(f) = pole(f) + arch(f) (zeta's pole and Gamma factor, NO primes) for explicit odd trial functions.
arch(f*f~) = -c_inf |f|^2 + int_0^inf w(u) D(u) du, D(u) = int |f(x+u) - f(x)|^2 dx = 2(g(0) - g(u)), w(u) = e^(u/2)/(2 sinh u)
(R2-curv Theorem E), pole = -2 <f, sinh(x/2)>^2 on odd f.   usage: python3 bg.py.txt"""
import mpmath as mp, importlib.util, importlib.machinery, flint
mp.mp.dps = 30
cinf = mp.log(mp.pi) - mp.digamma(mp.mpf(1) / 4)
w = lambda u: mp.exp(u / 2) / (2 * mp.sinh(u))
def Qbg(L, a):
    """f = sin(a x) on [-L, L]; returns pole, arch, total, norm^2, g"""
    L = mp.mpf(L)
    f = lambda x: mp.sin(a * x) if abs(x) <= L else mp.mpf(0)
    g = lambda u: mp.quad(lambda x: mp.sin(a * x) * mp.sin(a * (x - u)), [-L + u, L]) if u < 2 * L else mp.mpf(0)
    n2 = g(mp.mpf(0))
    S = mp.quad(lambda x: mp.sin(a * x) * mp.sinh(x / 2), [-L, L])
    pole = -2 * S ** 2
    I1 = mp.quad(lambda u: w(u) * 2 * (n2 - g(u)), [0, L / 4, L, 2 * L])
    I2 = 2 * n2 * mp.quad(w, [2 * L, 4, mp.inf])
    arch = -cinf * n2 + I1 + I2
    return pole, arch, pole + arch, n2, g
print("c_inf =", mp.nstr(cinf, 12))
