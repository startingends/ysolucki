import os, subprocess, sys

sys.path.insert(0, os.path.join(ROOT, "tools")); import records
TH = os.path.join(ROOT, "rh", "THEORY.md")
def run(*a):
    p = subprocess.run([sys.executable, os.path.join(ROOT, "tools", "records.py"), "theory"] + list(a), capture_output=True, text=True)
    if p.returncode: sys.exit(f"FAILED {a[:3]}: {p.stdout}{p.stderr}")
    return p.stdout.strip()
def get(tag, field):
    _, items = records.parse("theory", TH)
    return next(x for x in items if x["n"] == int(tag[1:]))["f"].get(field, "")
def app(tag, field, text, sep=" | "):
    old = get(tag, field); run("set", tag, field, (old + sep if old else "") + text)
def add(name, st, dep="", proof="", src="", routes="", note="", cb="", status=None, why=""):
    a = ["add", "--name", name, "--statement", st]
    if dep: a += ["--depends", dep]
    if proof and status in ("proved", "checked"): a += ["--proof", proof]
    if src: a += ["--source", src]
    if routes: a += ["--routes", routes]
    t = run(*a).split()[-1]
    if proof and status not in ("proved", "checked"): run("set", t, "proof", proof)
    if note: run("set", t, "note", note)
    if cb: run("set", t, "checked by", cb)
    if status in ("checked", "known", "refuted"): run("status", t, status, why or "see note")
    print(t, status or "open", name[:70]); return t
