import os, subprocess, sys
ROOT = os.path.abspath(os.path.join(os.path.dirname(os.path.abspath(__file__)), "..", "..", ".."))
sys.path.insert(0, os.path.join(ROOT, "tools")); import records
TH = os.path.join(ROOT, "rh", "THEORY.md")
def run(*a):
    p = subprocess.run([sys.executable, os.path.join(ROOT, "tools", "records.py"), "theory"] + list(a), capture_output=True, text=True)
    if p.returncode: sys.exit(f"FAILED {a[:3]}: {p.stdout}{p.stderr}")
    return p.stdout.strip()
def get(tag, field):
    _, items = records.parse("theory", TH)
    return next(x for x in items if x["n"] == int(tag[1:]))["f"].get(field, "")
def app(tag, field, text, sep=" | "):
    old = get(tag, field); run("set", tag, field, (old + sep if old else "") + text)
def add(name, statement, depends="", proof="", source="", routes="", note="", checked_by="", status=None, why=""):
    a = ["add", "--name", name, "--statement", statement]
    if depends: a += ["--depends", depends]
    if proof and status in ("proved", "checked"): a += ["--proof", proof]
    if source: a += ["--source", source]
    if routes: a += ["--routes", routes]
    tag = run(*a).split()[-1]
    if proof and status not in ("proved", "checked"): run("set", tag, "proof", proof)
    if note: run("set", tag, "note", note)
    if checked_by: run("set", tag, "checked by", checked_by)
    if status in ("checked", "known", "formal", "refuted"): run("status", tag, status, why or "see note")
    print(tag, status or "open", name[:70]); return tag
