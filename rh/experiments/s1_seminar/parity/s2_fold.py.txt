"""S1-parity, seminar S2 (answer to B6): the folding identity q(f_e) = 2[T(phi) + H(phi)], q(f_o) = 2[T(phi) - H(phi)] read as a split of the
prime sum at sqrt(D) = e^L, evaluated on the lowest EVEN eigenvector (folded phi on (0, L]) of (i) the impostor at L = 1.72 (its failure),
(ii) zeta at L = 1.0 and 1.5. H = Hankel (mirror) form of the full kernel on (0, L]; T = Weil form of phi alone (window of half size).
  zeta:      H = E_+^2 - sum_{m>=1} E_m^2 - H_Lambda,   E_+ = int phi e^{x/2}, E_m = int phi e^{-(2m+1/2)x}
  impostor:  H = - sum_{m>=0} E'_m^2 - H_c,           E'_m = int phi e^{-(2m+3/2)x}  (no pole; Gamma((s+1)/2))
  H_c = sum_n c_n n^{-1/2} (phi*phi)(log n).   T is obtained as mu/2 - H (mu = eigenvalue of the unit vector, |f_e| = 1).
usage: python3 s2_fold.py.txt check | run"""

def dh_c(X):
    kap = (mp.sqrt(10 - 2 * mp.sqrt(5)) - 2) / (mp.sqrt(5) - 1)
    a = [mp.mpf(0)] + [(mp.mpf(0), mp.mpf(1), kap, -kap, mp.mpf(-1))[n % 5] for n in range(1, X + 1)]
    b = list(a); b[1] = mp.mpf(0); l = [mp.mpf(0)] * (X + 1); pw = list(b); k = 1
    while 2 ** k <= X:
        for n in range(2, X + 1): l[n] += mp.mpf((-1) ** (k + 1)) / k * pw[n]
        new = [mp.mpf(0)] * (X + 1)
        for d in range(2, X + 1):
            if pw[d] != 0:
                for e in range(2, X // d + 1): new[d * e] += pw[d] * b[e]
        pw = new; k += 1
    return {n: l[n] * mp.log(n) for n in range(2, X + 1) if l[n] != 0}
def coeffs(kind, L):
    X = int(math.exp(2 * L) + 1e-9)
    if kind == 'zeta': return {n: c for n, c in wb.mangoldt(X).items() if math.log(n) < 2 * L - 1e-12}
    return {n: c for n, c in dh_c(X).items() if math.log(n) < 2 * L - 1e-12}
def blocks(kind, L, N, lat='int'):
    Lm = mp.mpf(L)
    if kind == 'zeta':
        M = wb.add(wb.arch_blocks(Lm, N, lat), wb.pole_blocks(Lm, N, lat))
    else:
        w = wb.freqs(Lm, N, lat); vals = []
        for om in w:

        M = wb.blocks(Lm, N, lat, [v[0] for v in vals], [v[1] for v in vals]); M = (M[0].apply(mp.re), M[1].apply(mp.re))
    for n, c in coeffs(kind, L).items(): M = wb.add(M, wb.atom_blocks(Lm, N, lat, n), c)
    return M
def hankel_parts(kind, L, phi, a_int=None):
    """phi: callable on (0, L]. Returns dict of pieces of H(phi)."""
    Lm = mp.mpf(L)
    E = lambda a: mp.quad(lambda x: phi(x) * mp.exp(a * x), mp.linspace(0, Lm, 9))
    conv = lambda u: mp.quad(lambda x: phi(x) * phi(u - x), mp.linspace(max(mp.mpf(0), u - Lm), min(Lm, u), 9))
    out = {}
    if kind == 'zeta':
        out['pole E_+^2'] = E(mp.mpf(1) / 2) ** 2
        gam = mp.fsum(E(-(2 * m + mp.mpf(1) / 2)) ** 2 for m in range(1, 400)); out['Gamma squares'] = gam
        out['H_arith'] = {n: c / mp.sqrt(n) * conv(mp.log(n)) for n, c in coeffs(kind, L).items()}
        out['H'] = out['pole E_+^2'] - gam - mp.fsum(out['H_arith'].values())
    else:
        out['pole E_+^2'] = mp.mpf(0)
        gam = mp.fsum(E(-(2 * m + mp.mpf(3) / 2)) ** 2 for m in range(0, 400)); out['Gamma squares'] = gam
        out['H_arith'] = {n: c / mp.sqrt(n) * conv(mp.log(n)) for n, c in coeffs(kind, L).items()}
        out['H'] = - gam - mp.fsum(out['H_arith'].values())
    return out
