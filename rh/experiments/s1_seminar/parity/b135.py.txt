"""S1-parity, answer to board B135 (SP-operator): t_1(L) = <1, Q_L^-1 1> (1 = the constant function on the window) for zeta and for the
Davenport-Heilbronn impostor, and one-sided slopes of log t_1 around L_n = (1/2) log n, n = 2, 3, 4.
Integer lattice (the constant is the k = 0 basis vector times sqrt(2L)):  t_1 = 2L (Q_even^-1)_00.
usage: python3 b135.py.txt grid | jump <N>"""
import sys, math, time, importlib.util, importlib.machinery
import mpmath as mp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    m = importlib.util.module_from_spec(spec); spec.loader.exec_module(m); return m
fd = load("fd", "/home/user/ysolucki/rh/experiments/s1_seminar/parity/fd.py.txt"); wb = fd.wb
t0 = time.time(); ns = lambda x, k=6: mp.nstr(x, k)
def dh_c(X):
    kap = (mp.sqrt(10 - 2 * mp.sqrt(5)) - 2) / (mp.sqrt(5) - 1)
    a = [mp.mpf(0)] + [(mp.mpf(0), mp.mpf(1), kap, -kap, mp.mpf(-1))[n % 5] for n in range(1, X + 1)]
    b = list(a); b[1] = mp.mpf(0); l = [mp.mpf(0)] * (X + 1); pw = list(b); k = 1
    while 2 ** k <= X:
        for n in range(2, X + 1): l[n] += mp.mpf((-1) ** (k + 1)) / k * pw[n]
        new = [mp.mpf(0)] * (X + 1)
        for d in range(2, X + 1):
            if pw[d] != 0:
                for e in range(2, X // d + 1): new[d * e] += pw[d] * b[e]
        pw = new; k += 1
    return {n: l[n] * mp.log(n) for n in range(2, X + 1) if l[n] != 0}
def even_block(kind, L, N):
    if kind == 'zeta':
        M = wb.add(wb.arch_blocks(L, N, 'int'), wb.pole_blocks(L, N, 'int'))
        at = wb.mangoldt(math.exp(2 * L))
    else:
        Lm = mp.mpf(L); w = wb.freqs(Lm, N, 'int'); vals = []
        for om in w:
            v, dv = fd.phi_arch_c(om + fd.I, Lm); vals.append((v + mp.log(5) / 2, dv))
        M = wb.blocks(L, N, 'int', [v[0] for v in vals], [v[1] for v in vals]); M = (M[0].apply(mp.re), M[1].apply(mp.re))
        at = dh_c(max(2, int(math.exp(2 * L)))) if math.exp(2 * L) >= 2 else {}
    for x, c in at.items():
        if math.log(float(x)) < 2 * L - 1e-12: M = wb.add(M, wb.atom_blocks(L, N, 'int', x), c)
    return M[0]
def t1(kind, L, N):
    E = even_block(kind, L, N); n = E.rows
    e0 = mp.matrix(n, 1); e0[0] = 1
    x = mp.lu_solve(E, e0)
    return 2 * L * x[0], min(mp.eigsy(E, eigvals_only=True))
