"""S1-parity, plank zero of Hunch 1 (rh/HUNCHES.md): size of the cut Moebius-type operator M_l = Z_l^-1 on a window.
(C_l f)(v) = sum_{log m <= v} c(m) m^-1/2 f(v - log m), v in [0, l], l = 2L.  Floats (numpy); definitions in HUNCH1.md section 0.
usage: python3 h1.py.txt table | identity | longrange"""
import sys, math, time
import numpy as np
t0 = time.time()
KAP = (math.sqrt(10 - 2 * math.sqrt(5)) - 2) / (math.sqrt(5) - 1)
def dinv(a, X):
    """Dirichlet inverse b of a (a[1] = 1), arrays indexed 0..X"""
    b = np.zeros(X + 1); b[1] = 1.0
    for d in range(1, X + 1):
        if b[d] == 0: continue
        # b[n] -= a[n/d] b[d] for n = d*k, k >= 2
        ks = np.arange(2, X // d + 1)
        if len(ks): b[d * ks] -= a[ks] * b[d]
    return b
def dconv(f, g, X):
    h = np.zeros(X + 1)
    for d in range(1, X + 1):
        if f[d] == 0: continue
        ks = np.arange(1, X // d + 1); h[d * ks] += f[d] * g[ks]
    return h
def systems(X, d=0.25):
    n = np.arange(X + 1, dtype=float)
    one = np.ones(X + 1); one[0] = 0
    mu = dinv(one, X)
    adh = np.array([0.0] + [(0.0, 1.0, KAP, -KAP, -1.0)[k % 5] for k in range(1, X + 1)])
    bdh = dinv(adh, X)
    with np.errstate(divide='ignore'):
        up = np.where(n > 0, mu * n ** d, 0.0); dn = np.where(n > 0, mu * np.where(n > 0, n, 1.0) ** (-d), 0.0)
    fd = dconv(up, dn, X); f0 = dconv(mu, mu, X)
    op = np.zeros(X + 1); op[1] = 1.0
    if X >= 2: op[2] = -1.0
    return {"zeta": mu, "impostor": bdh, "F_d (d=1/4)": fd, "zeta^2 (d=0)": f0, "one-prime": op}

def grid_norm(c, l, n=6000, absolute=False, iters=400):
    """operator norm of C_l on L^2[0,l] in a grid of n cells with shifts rounded to the grid (power iteration on C^T C)."""
    h = l / n; X = int(math.floor(math.exp(l) + 1e-12))
    terms = [(int(round(math.log(m) / h)), (abs(c[m]) if absolute else c[m]) / math.sqrt(m)) for m in range(1, X + 1) if c[m] != 0]
    def ap(f):
        g = np.zeros(n)
        for k, w in terms:
            if k < n: g[k:] += w * f[:n - k]
        return g
    def apT(g):
        f = np.zeros(n)
        for k, w in terms:
            if k < n: f[:n - k] += w * g[k:]
        return f
    rng = np.random.default_rng(0); f = rng.normal(size=n); f /= np.linalg.norm(f); val = 0
    for _ in range(iters):
        g = ap(f); f = apT(g); nv = np.linalg.norm(f)
        if nv == 0: return 0.0
        f /= nv; val = math.sqrt(nv)
    return val
def GK(c, l, K, n=8000):
    """sup |C_l f|/|f| over trigonometric polynomials of degree <= K/2 on [0, l]; exact pointwise evaluation, midpoint quadrature."""
    h = l / n; v = (np.arange(n) + 0.5) * h; X = int(math.floor(math.exp(l) + 1e-12))
    ks = np.arange(-(K // 2), K // 2 + 1); om = 2 * np.pi * ks / l
    Mf = np.zeros((n, len(ks)), dtype=complex)
    for m in range(1, X + 1):
        if c[m] == 0: continue
        lm = math.log(m); mask = (v >= lm)[:, None]
        Mf += (c[m] / math.sqrt(m)) * mask * np.exp(1j * np.outer(v - lm, om))
    Gm = (Mf.conj().T @ Mf) * h / l                      # modes e^{i om v}/sqrt(l) are orthonormal
    return math.sqrt(max(np.linalg.eigvalsh(Gm)))

    def H(c, f): return lambda x: sum(c[m] / math.sqrt(m) * f(math.log(m) - x) for m in range(1, X + 1) if c[m] != 0 and 0 < math.log(m) - x <= l)
