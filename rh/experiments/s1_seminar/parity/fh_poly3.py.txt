def base(par):
    idx = [j for j in range(NB) if j % 2 == par]
    A, G, _ = mats(par)
    cfs = {(j, k): [tomp(c) for c in cross(j, k)] for j in idx for k in idx if k >= j}
    Lc = mp.cholesky(G); Li = mp.inverse(Lc)
    return idx, Li * A * Li.T, cfs, Li
B = [base(0), base(1)]
def interval(y):
    """set of m >= 0 with both blocks >= 0, or None. A is indefinite, so: block >= 0 on {m} is an interval; find it from the pencil of (An, Cn)."""
    lo, hi = mp.mpf(0), mp.inf
    for idx, An, cfs, Li in B:
        n = len(idx); C = mp.matrix(n, n)
        for a, j in enumerate(idx):
            for b, k in enumerate(idx):
                cf = cfs[(min(j, k), max(j, k))]
                C[a, b] = mp.fsum(cf[nn] * y ** nn for nn in range(len(cf)))
        Cn = Li * C * Li.T
        f = lambda m: min(mp.eigsy(An - 2 * m * Cn, eigvals_only=True))
        # concave in m: maximise, then bisect the two ends
        a_, b_ = mp.mpf('0.3'), mp.mpf('0.7'); g = (mp.sqrt(5) - 1) / 2
        x1, x2 = b_ - g * (b_ - a_), a_ + g * (b_ - a_); f1, f2 = f(x1), f(x2)
        for _ in range(60):
            if f1 < f2: a_, x1, f1 = x1, x2, f2; x2 = a_ + g * (b_ - a_); f2 = f(x2)
            else: b_, x2, f2 = x2, x1, f1; x1 = b_ - g * (b_ - a_); f1 = f(x1)
        ms = (a_ + b_) / 2
        if f(ms) < 0: return None
        e = []
        for sgn in (-1, 1):
            p, q = ms, ms + sgn * mp.mpf('0.2')
            for _ in range(60):
                mid = (p + q) / 2
                if f(mid) >= 0: p = mid
                else: q = mid
            e.append(p)
        lo, hi = max(lo, e[0]), min(hi, e[1])
    return (lo, hi) if lo <= hi else None
