G_T(L) := max over |t| <= T of s_L(t): a lower bound for the norm on any class containing the frequencies up to T."""

Ls = (0.5, 1.0, 1.5, 1.72, 2.0, 2.5, 3.0, 3.5, 4.0, 5.0, 6.0)
X = int(math.exp(2 * max(Ls))) + 1; S = systems(X)
ts = np.arange(0, 100.0001, 0.02)
n = np.arange(1, X + 1, dtype=float); ln = np.log(n)
for name in ("zeta", "impostor", "F_d (d=1/4)", "zeta^2 (d=0)", "one-prime"):
    c = S[name][1:]
    out = {T: [] for T in (4, 10, 100)}; arg = []
    for L in Ls:
        l = 2 * L; y = int(math.floor(math.exp(l) + 1e-12))
        w = np.diff(np.concatenate([ln[:y], [l]]))                       # lengths of the v-intervals on which the partial sum is constant
        # partial sums for all t at once, in chunks
        s2 = np.zeros(len(ts))
        ch = max(1, int(3e6 / y))
        for a in range(0, len(ts), ch):
            tt = ts[a:a + ch]
            ph = (c[:y] * n[:y] ** -0.5)[None, :] * np.exp(-1j * np.outer(tt, ln[:y]))
            D = np.cumsum(ph, axis=1)
            s2[a:a + ch] = (np.abs(D) ** 2 * w[None, :]).sum(axis=1) / l
        s = np.sqrt(s2)
        for T in out: out[T].append(s[ts <= T].max())
        arg.append(ts[np.argmax(s)])
    for T in (4, 10, 100):
