"""S1-parity: two extra mirror matrices, same route as mirror2.py.txt (Dirichlet logarithm, exact rationals, 60 digits).
 (i)  F = L(s,chi) L(s,chi-bar), chi mod 5 with chi(2) = i  (Socrates' control, board B6)
 (ii) F = zeta(s) zeta(s-1) = sum sigma(n) n^-s            (S1-parity's control: mirror positive, zeros off the centre line Re s = 1)
 (iii) dial n = 4 = 8 (1 - 4^(1-s)) zeta(s) zeta(s-1): structure of the single negative eigenvalue
usage: python3 mirror3.py.txt [A]"""
import sys
from fractions import Fraction as Fr
import mpmath as mp
mp.mp.dps = 60
A = int(sys.argv[1]) if len(sys.argv) > 1 else 12
X = A * A
def dconv(f, g, zero):
    h = [zero] * (X + 1)
    for d in range(1, X + 1):
        if f[d] == 0: continue
        for e in range(1, X // d + 1):
            if g[e] != 0: h[d * e] += f[d] * g[e]
    return h
def dlog(a):
    b = list(a); b[1] = Fr(0); l = [Fr(0)] * (X + 1); pw = list(b); k = 1
    while 2 ** k <= X:
        for n in range(2, X + 1): l[n] += Fr((-1) ** (k + 1), k) * pw[n]
        pw = dconv(pw, b, Fr(0)); k += 1
    return l
def cfrom(l): return [mp.mpf(0), mp.mpf(0)] + [mp.mpf(l[n].numerator) / l[n].denominator * mp.log(n) for n in range(2, X + 1)]
def report(name, c):
    M = mp.matrix(A - 1, A - 1)
    for a in range(2, A + 1):
        for b in range(2, A + 1): M[a - 2, b - 2] = c[a * b]
    ev = sorted(mp.eigsy(M, eigvals_only=True)); sc = max(abs(e) for e in ev); tol = mp.mpf(10) ** (-40) * sc
    print("%-34s a,b=2..%d: min eig %s ; #neg %d ; #pos %d ; c(4) = %s ; c_2..c_9 = %s" % (name, A, mp.nstr(ev[0], 8), sum(1 for e in ev if e < -tol),
          sum(1 for e in ev if e > tol), mp.nstr(c[4], 8), " ".join(mp.nstr(c[k], 4) for k in range(2, 10))))
    return ev
chi = {0: 0, 1: 1, 2: 1j, 3: -1j, 4: -1}
x1 = [0] + [chi[n % 5] for n in range(1, X + 1)]; x2 = [0] + [chi[n % 5].conjugate() if n % 5 else 0 for n in range(1, X + 1)]
prod = dconv(x1, x2, 0)
assert all(abs(complex(v).imag) < 1e-9 for v in prod)
aLL = [Fr(0)] + [Fr(int(round(complex(v).real))) for v in prod[1:]]
print("L(chi)L(chi-bar): a_1..a_12 =", [str(v) for v in aLL[1:13]])
report("L(chi) L(chi-bar), chi mod 5", cfrom(dlog(aLL)))
sig = [Fr(0)] + [Fr(sum(d for d in range(1, n + 1) if n % d == 0)) for n in range(1, X + 1)]
report("zeta(s) zeta(s-1)  (sigma)", cfrom(dlog(sig)))
r4 = [Fr(0)] + [Fr(sum(d for d in range(1, n + 1) if n % d == 0 and d % 4)) for n in range(1, X + 1)]
c4 = cfrom(dlog(r4)); ev = report("dial n=4  (Jacobi r_4/8)", c4)
print("  dial n=4 at powers of 2: c(2^j)/log 2, j=1..6 = %s   [claim: 1 - (-2)^j]" % [mp.nstr(c4[2 ** j] / mp.log(2), 6) for j in range(1, 7) if 2 ** j <= X])
print("  dial n=4 at powers of 3: c(3^j)/log 3, j=1..4 = %s   [claim: 1 + 3^j]" % [mp.nstr(c4[3 ** j] / mp.log(3), 6) for j in range(1, 5) if 3 ** j <= X])
