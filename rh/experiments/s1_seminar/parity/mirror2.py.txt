"""S1-parity: independent recomputation of the lead's mirror-matrix table (idea #21, e21_reflection/mirror.log).
Second route, nothing imported from the lead's script or from r1_A:
  * coefficients of -F'/F through the Dirichlet-series LOGARITHM  l = sum_k (-1)^(k+1)/k (a - delta)^{*k},  c_n = l_n log n
    (the lead solves the recursion a_m log m = sum_{d|m} c_d a_{m/d});
  * theta dial: theta^n = (1+u)^n = sum_k binom(n,k) u^k, u = 2 sum q^(j^2), in EXACT rationals (n = 7/5, 4, ...)
    (the lead uses the power-of-a-series recurrence in floats); n = 4 also by Jacobi r_4(m) = 8 sum_{d|m, 4 !| d} d
    and by the closed form c_m = (1+m) Lambda(m) - [m = 4^k] m log 4;
  * impostor: a_n = (1, kappa, -kappa, -1, 0) mod 5 at 60 digits; c_n validated against the FUNCTION: -f'/f(s) at s = 12, 20
    computed from Hurwitz zeta values (tools/controls.py is not used either: own formula);
  * eigenvalues with mpmath at 60 digits (lead: numpy double).
usage: python3 mirror2.py.txt [A]     (matrix on a, b = 2..A; default 12)"""
import sys, time
from fractions import Fraction as Fr
import mpmath as mp
mp.mp.dps = 60
A = int(sys.argv[1]) if len(sys.argv) > 1 else 12
X = A * A
t0 = time.time()

def dconv(f, g, zero):
    h = [zero] * (X + 1)
    for d in range(1, X + 1):
        if f[d] == 0: continue
        for e in range(1, X // d + 1):
            if g[e] != 0: h[d * e] += f[d] * g[e]
    return h

def dlog(a, zero, one):
    """l with sum l_n n^-s = log sum a_n n^-s, a_1 = 1 (exact in whatever ring a lives in, apart from 1/k)."""
    b = list(a); b[1] = zero                      # a - delta, supported on n >= 2
    l = [zero] * (X + 1); pw = list(b); k = 1
    while 2 ** k <= X:
        for n in range(2, X + 1):
            if pw[n] != 0: l[n] += (one * (-1) ** (k + 1) / k) * pw[n]
        pw = dconv(pw, b, zero); k += 1
    return l

def cfrom(l):
    return [mp.mpf(0), mp.mpf(0)] + [(mp.mpf(l[n].numerator) / l[n].denominator if isinstance(l[n], Fr) else l[n]) * mp.log(n) for n in range(2, X + 1)]

def mangoldt(n):
    for p in range(2, n + 1):
        if n % p == 0:
            while n % p == 0: n //= p
            return mp.log(p) if n == 1 else mp.mpf(0)
    return mp.mpf(0)

def chi4(n): return 0 if n % 2 == 0 else (1 if n % 4 == 1 else -1)

def dial_exact(nfr):
    """a_m = r_n(m)/(2n), exact rational, via the binomial series."""
    u = [0] * (X + 1); j = 1
    while j * j <= X: u[j * j] = 2; j += 1
    th = [Fr(0)] * (X + 1); th[0] = Fr(1)
    pw = [0] * (X + 1); pw[0] = 1; binom = Fr(1)
    for k in range(1, X + 1):
        new = [0] * (X + 1)
        for i in range(X + 1):
            if pw[i]:
                for jj in range(1, int((X - i) ** 0.5) + 1):
                    new[i + jj * jj] += pw[i] * 2
        pw = new
        binom = binom * (nfr - (k - 1)) / k
        if binom == 0: break
        for i in range(k, X + 1):
            if pw[i]: th[i] += binom * pw[i]
    return [Fr(0)] + [th[m] / (2 * nfr) for m in range(1, X + 1)]

cases = {}
# zeta, L(chi_4): coefficients a_n = 1, chi_4(n): through the logarithm route too (not typed in as Lambda)
one = [Fr(0)] + [Fr(1)] * X
cases["zeta"] = cfrom(dlog(one, Fr(0), Fr(1)))
cases["L(chi_4)"] = cfrom(dlog([Fr(0)] + [Fr(chi4(n)) for n in range(1, X + 1)], Fr(0), Fr(1)))
# impostor
kap = (mp.sqrt(10 - 2 * mp.sqrt(5)) - 2) / (mp.sqrt(5) - 1)
adh = [mp.mpf(0)] + [(mp.mpf(0), mp.mpf(1), kap, -kap, mp.mpf(-1))[n % 5] for n in range(1, X + 1)]
ldh = dlog(adh, mp.mpf(0), mp.mpf(1))
cases["impostor (DH)"] = [mp.mpf(0), mp.mpf(0)] + [ldh[n] * mp.log(n) for n in range(2, X + 1)]
d14 = dial_exact(Fr(7, 5)); d4 = dial_exact(Fr(4)); d1 = dial_exact(Fr(1)); d2 = dial_exact(Fr(2))
cases["theta dial n=1.4"] = cfrom(dlog(d14, Fr(0), Fr(1)))
cases["theta dial n=4"] = cfrom(dlog(d4, Fr(0), Fr(1)))
cases["theta dial n=1"] = cfrom(dlog(d1, Fr(0), Fr(1)))
cases["theta dial n=2"] = cfrom(dlog(d2, Fr(0), Fr(1)))

print("checks of the coefficients (A = %d, X = %d):" % (A, X))
print("  zeta: max |c_n - Lambda(n)| = %s" % mp.nstr(max(abs(cases["zeta"][n] - mangoldt(n)) for n in range(2, X + 1)), 3))
print("  L(chi_4): max |c_n - chi_4(n) Lambda(n)| = %s" % mp.nstr(max(abs(cases["L(chi_4)"][n] - chi4(n) * mangoldt(n)) for n in range(2, X + 1)), 3))
jac = all(d4[m] * 8 == 8 * sum(d for d in range(1, m + 1) if m % d == 0 and d % 4) for m in range(1, X + 1))
print("  dial n=4: binomial-series r_4(m) == Jacobi 8*sum_{d|m, 4 not| d} d for all m <= X: %s" % jac)
def c4closed(m):
    v = (1 + m) * mangoldt(m); k = m
    while k % 4 == 0: k //= 4
    return v - (m * mp.log(4) if (k == 1 and m > 1) else 0)
print("  dial n=4: max |c_m - [(1+m)Lambda(m) - [m=4^k] m log 4]| = %s" % mp.nstr(max(abs(cases["theta dial n=4"][m] - c4closed(m)) for m in range(2, X + 1)), 3))
sq = lambda n: int(round(n ** 0.5)) ** 2 == n
print("  dial n=1: max |c_m - 2 Lambda(sqrt m)[m square]| = %s" % mp.nstr(max(abs(cases["theta dial n=1"][m] - (2 * mangoldt(int(round(m ** 0.5))) if sq(m) else 0)) for m in range(2, X + 1)), 3))
print("  dial n=2: max |c_m - (1+chi_4(m))Lambda(m)| = %s" % mp.nstr(max(abs(cases["theta dial n=2"][m] - (1 + chi4(m)) * mangoldt(m)) for m in range(2, X + 1)), 3))
print("  dial n=7/5: r(1..8)/(2n) exact = %s" % [str(x) for x in d14[1:9]])
# impostor against the function itself
def dh(s): return mp.power(5, -s) * mp.fsum(adh[r] * mp.zeta(s, mp.mpf(r) / 5) for r in (1, 2, 3, 4))
def dhp(s): return -mp.log(5) * dh(s) + mp.power(5, -s) * mp.fsum(adh[r] * mp.zeta(s, mp.mpf(r) / 5, 1) for r in (1, 2, 3, 4))
for s in (12, 20):
    lhs = -dhp(mp.mpf(s)) / dh(mp.mpf(s)); rhs = mp.fsum(cases["impostor (DH)"][n] * mp.power(n, -s) for n in range(2, X + 1))
    print("  impostor: -f'/f(%d) from Hurwitz zeta = %s ; sum_{n<=X} c_n n^-s = %s ; diff %s (tail ~ %s)" % (s, mp.nstr(lhs, 20), mp.nstr(rhs, 20),
          mp.nstr(abs(lhs - rhs), 3), mp.nstr(mp.power(X + 1, -s) * 5, 3)))

print()
print("%-20s %14s %5s %5s %5s   %14s   %s" % ("function", "min eig [c(ab)]", "rank", "#neg", "#pos", "min eig [c(ab)/sqrt(ab)]", "c_2..c_9"))
tol = mp.mpf(10) ** (-40)
res = {}
for name in ("zeta", "impostor (DH)", "L(chi_4)", "theta dial n=1.4", "theta dial n=4", "theta dial n=1", "theta dial n=2"):
    c = cases[name]
    M = mp.matrix(A - 1, A - 1); Ms = mp.matrix(A - 1, A - 1)
    for a in range(2, A + 1):
        for b in range(2, A + 1):
            M[a - 2, b - 2] = c[a * b]; Ms[a - 2, b - 2] = c[a * b] / mp.sqrt(a * b)
    ev = sorted(mp.eigsy(M, eigvals_only=True)); evs = sorted(mp.eigsy(Ms, eigvals_only=True))
    scale = max(abs(e) for e in ev)
    neg = sum(1 for e in ev if e < -tol * scale); pos = sum(1 for e in ev if e > tol * scale)
    res[name] = ev
    print("%-20s %14s %5d %5d %5d   %14s   %s" % (name, mp.nstr(ev[0], 8), neg + pos, neg, pos, mp.nstr(evs[0], 8),
          " ".join(mp.nstr(c[k], 4) for k in range(2, 10))))
print()
for name in ("impostor (DH)", "theta dial n=1.4", "theta dial n=4"):
    print("  %s negative eigenvalues: %s" % (name, [mp.nstr(e, 8) for e in res[name] if e < -tol]))
print("total %.1fs" % (time.time() - t0))
