Kernel of F_d = zeta's kernel times m(u) = 2cosh(d u)  (S1-mirror B15):  Q_d(g) = Q(m g), g = f * f~.
  primes: Lambda(n) n^{-1/2} m(log n);  arch: 2 c0 g(0) + int_0^inf ( 2 g(0) e^{-2u}/u - 2 w(u) m(u) g(u) ) du;
  pole: 2cosh((1/2+d)u) + 2cosh((1/2-d)u): four functionals e^{+-(1/2+-d)x}.
Identity to test (derived by hand, NOTES.md):  Q_d(f) = 2 Q(cosh(dx) f) - 2 Q(sinh(dx) f).
usage: fd.py.txt <what> ..."""
import sys, math, importlib.util, importlib.machinery, numpy as np, mpmath as mp
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm
box = load("box", "/home/user/ysolucki/rh/experiments/s1_seminar/cubic/box.py.txt")
mp.mp.dps = 30
_cache = {}
def parts(L, n):
    """d-independent data"""
    key = (L, n)
    if key not in _cache:
        _cache[key] = (box.lam(math.exp(2 * L) + 1e-9),)
    return _cache[key]
def boxNd(L, n, d):
    """returns N_d (primes_d - arch_d), list of (p_sigma, q_sigma) for sigma = 1/2+d, 1/2-d"""
    h = mp.mpf(2) * L / n; c0 = -mp.log(mp.pi); cn = parts(L, n)[0]
    m = lambda u: 2 * mp.cosh(d * u)
    tri = lambda v: max(mp.mpf(0), h - abs(v))
    A = []
    for k in range(n):
        dist = k * h
        if k == 0:
            val = 2 * c0 * h + mp.quad(lambda u: 2 * h * mp.exp(-2 * u) / u - 2 * box.w(u) * m(u) * (h - u), [0, h / 2, h]) + 2 * h * mp.e1(2 * h)
        else:
            val = -mp.quad(lambda u: box.w(u) * m(u) * tri(u - dist), [dist - h, dist, dist + h])
        A.append(val / h)
    N = np.zeros((n, n))
    for i in range(n):
        for j in range(n):
            dist = (i - j) * h; pr = mp.mpf(0)
            for x, c in cn.items():
                u = mp.log(x)
                pr += c / mp.sqrt(x) * m(u) * (tri(u - dist) + tri(u + dist))
            N[i, j] = float(pr / h - A[abs(i - j)])
    xs = [-mp.mpf(L) + i * h for i in range(n + 1)]
    fun = []
    for sg in (mp.mpf(1) / 2 + d, mp.mpf(1) / 2 - d):
        p = np.array([float((mp.exp(sg * xs[i + 1]) - mp.exp(sg * xs[i])) / (sg * mp.sqrt(h))) for i in range(n)])
        q = np.array([float((mp.exp(-sg * xs[i]) - mp.exp(-sg * xs[i + 1])) / (sg * mp.sqrt(h))) for i in range(n)])
        fun.append((p, q))
    return N, fun
def Qd(L, n, d):
    N, fun = boxNd(L, n, d)
    P = sum(np.outer(p, q) + np.outer(q, p) for p, q in fun)
    return P - N, N, fun
def expker(kappa, L, n):
    """box compression of e^{kappa |x-y|} (kappa of either sign)"""
    h = 2 * L / n; dd = np.abs(np.subtract.outer(np.arange(n), np.arange(n)))
    M = (2 * math.sinh(kappa * h / 2) / kappa) ** 2 / h * np.exp(kappa * dd * h)
    np.fill_diagonal(M, 2 * (math.exp(kappa * h) - 1 - kappa * h) / (kappa ** 2 * h))
    return M
def blocks(Q):
    n = len(Q); J = np.eye(n)[::-1]; Pe = (np.eye(n) + J) / 2; Po = (np.eye(n) - J) / 2
    big = 10 * abs(Q).max()
    return np.linalg.eigvalsh(Pe @ Q @ Pe + big * Po)[0], np.linalg.eigvalsh(Po @ Q @ Po + big * Pe)[0]
