def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm

def be(H, a):
    m = len(a); s = H @ a; I = np.eye(m)
    Lg = (H * a[None, :]) / s[:, None] - I; r = Lg - np.diag(np.diag(Lg)); pi = a * s
    Gam = [0.5 * sum(r[i, j] * np.outer(I[j] - I[i], I[j] - I[i]) for j in range(m)) for i in range(m)]
    Gam2 = []
    for i in range(m):
        LG = sum(r[i, j] * (Gam[j] - Gam[i]) for j in range(m))
        X = sum(r[i, j] * np.outer(I[j] - I[i], Lg[j] - Lg[i]) for j in range(m))
        Gam2.append(0.5 * (LG - 0.5 * (X + X.T)))
    def ok(rho): return all(np.linalg.eigvalsh(Gam2[i] - rho * Gam[i])[0] > -1e-9 for i in range(m))
    lo, hi = -50.0, 5.0
    for _ in range(50):
        mid = (lo + hi) / 2
        if ok(mid): lo = mid
        else: hi = mid
    A1 = np.diag(1 / a)
    T = np.array([np.outer(H[:, k], H[:, k]) / s[k] - (pi[k] * A1 @ (Gam2[k] - Gam[k]) @ A1) / a[k] for k in range(m)])
    gap = np.sort(np.linalg.eigvals(-Lg).real)[1]
    return T, lo, gap
def formula(H, a):
    m = len(a); s = H @ a; D = np.diag(a / s); HDH = H @ D @ H
    T = np.zeros((m, m, m))
    for i in range(m):
        for j in range(m):
            for k in range(m):
                if len({i, j, k}) == 3:
                    T[i, j, k] = 0.5 * (H[i, j] * H[i, k] / s[i] + H[i, j] * H[j, k] / s[j] + H[i, k] * H[j, k] / s[k])
                elif i == j == k:
                    T[i, i, i] = 1.5 * H[i, i] ** 2 / s[i] - 0.75 * HDH[i, i] / a[i] + s[i] / (4 * a[i] ** 2)
                else:
                    u = i if (i == j or i == k) else j          # repeated index
                    v = [t for t in (i, j, k) if t != u][0]
                    T[i, j, k] = H[u, u] * H[u, v] / s[u] + H[u, v] ** 2 / (2 * s[v]) - HDH[u, v] / (4 * a[u])
    return T
