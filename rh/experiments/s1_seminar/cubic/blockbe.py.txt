"""S1-cubic, board B42: BLOCK-AVERAGED Bakry-Emery.  For a partition of the sites into blocks S:
   rho_S := sup{ rho : sum_{k in S} pi_k (Gamma_2 - rho Gamma)(f)(k) >= 0 for all f },   rho* := min_S rho_S.
Summing over blocks gives ||Lf||^2 >= rho* E(f), i.e. gap >= rho*.  Blocks = single sites: pointwise Bakry-Emery.  One block: the gap itself.
Mean-field value (resampling chain, PROVED in TURN1.md): rho_S = (1 + 3 theta)/(2 + 2 theta), theta = pi(S)/pi(all), for |S| >= 2.
Also board B49: E = kernel e^{|x-y|/2} on boxes, N' = N - W0 (W0 = kernel e^{-|x-y|/2}), block matrix [[N',E],[E,E]]."""

def gammas(H, a):
    m = len(a); s = H @ a; I = np.eye(m)
    Lg = (H * a[None, :]) / s[:, None] - I; r = Lg - np.diag(np.diag(Lg)); pi = a * s
    Gam = [0.5 * sum(r[i, j] * np.outer(I[j] - I[i], I[j] - I[i]) for j in range(m)) for i in range(m)]
    Gam2 = []
    for i in range(m):
        LG = sum(r[i, j] * (Gam[j] - Gam[i]) for j in range(m))
        X = sum(r[i, j] * np.outer(I[j] - I[i], Lg[j] - Lg[i]) for j in range(m))
        Gam2.append(0.5 * (LG - 0.5 * (X + X.T)))
    gap = np.sort(np.linalg.eigvals(-Lg).real)[1]
    return Gam, Gam2, pi, gap
def rho_block(Gam, Gam2, pi, S):
    A2 = sum(pi[k] * Gam2[k] for k in S); A1 = sum(pi[k] * Gam[k] for k in S)
    lo, hi = -50.0, 50.0
    for _ in range(60):
        mid = (lo + hi) / 2
        if np.linalg.eigvalsh(A2 - mid * A1)[0] > -1e-10 * np.abs(A2).max(): lo = mid
        else: hi = mid
    return lo
