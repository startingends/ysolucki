"""S1-cubic turn 2: the canonical TWO-block certificate from the cone tensor: H_+- = H/2 +- (l_+ l_+^T - l_- l_-^T)/s,
l_+- = H a_+-, a_+- = Perron vector restricted to {x>0, F1} / {x<0, F2}.  Check: H_+ + H_- = H, H_+ a_- = H_- a_+, #pos(H_+-) <= 1."""
import math, importlib.util, importlib.machinery, numpy as np
def load(name, P):
    spec = importlib.util.spec_from_file_location(name, P, loader=importlib.machinery.SourceFileLoader(name, P))
    mm = importlib.util.module_from_spec(spec); spec.loader.exec_module(mm); return mm
box = load("box", "/home/user/ysolucki/rh/experiments/s1_seminar/cubic/box.py.txt")
for kind, L, n in (("zeta", 1.0, 24), ("zeta", 0.5, 24), ("beurling", 1.0, 24)):
    cn = box.lam(math.exp(2 * L)) if kind == "zeta" else box.beurling(math.exp(2 * L))
    Nb, p, q = box.boxN(L, n, cn); H = box.ext(Nb, p, q); ev, V = np.linalg.eigh(H); a = np.abs(V[:, -1]); m = n + 2
    ap = np.zeros(m); ap[n // 2:n] = a[n // 2:n]; ap[n] = a[n]; am = a - ap
    l = H @ a; s = a @ l; lp, lm = H @ ap, H @ am
    Hp = H / 2 + (np.outer(lp, lp) - np.outer(lm, lm)) / s; Hm = H - Hp
    ep = np.linalg.eigvalsh(Hp)
    print("%s L=%.1f n=%d: #pos(H)=%d ; |H_+ a_- - H_- a_+| = %.1e ; eig(H_+) top two %.4f %.3e ; #pos(H_+) = %d, #pos(H_-) = %d" % (
        kind, L, n, (ev > 1e-10).sum(), np.abs(Hp @ am - Hm @ ap).max(), ep[-1], ep[-2], (ep > 1e-10).sum(), (np.linalg.eigvalsh(Hm) > 1e-10).sum()))
