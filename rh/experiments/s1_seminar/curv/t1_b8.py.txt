"""S1-curv turn 1, answer to B8: is the integrated open-path term the Hankel (mirror) form of D5?  usage: t1_b8.py.txt L M kind
A = matrix of eta -> Op(h_-) - Op(h_+);  B = matrix of eta -> E(h_-) - E(h_+) = 4 Hankel_K(u eta); also edge-mirror variants."""
import sys, importlib.machinery, importlib.util, numpy as np
ld = importlib.machinery.SourceFileLoader('s1lib', 's1lib.py.txt'); sp = importlib.util.spec_from_loader('s1lib', ld); S = importlib.util.module_from_spec(sp); ld.exec_module(S)
L = float(sys.argv[1]); M = int(sys.argv[2]); kind = sys.argv[3]
b = S.build(L, M, kind); ev, lam1, u, q, Lg = S.process(b['N']); pi = u * u; k = b['k']; h = M // 2
E = np.eye(M); inW = lambda j: 0 <= j < M
Op = np.zeros((M, M))
for y in range(M):
    for a in range(-M + 1, M):
        if a == 0 or not inW(y + a): continue
        ga = k[abs(a)] * u[y + a] * (E[y + a] - E[y])
        for bb in range(-M + 1, M):
            if bb == 0 or not inW(y + bb) or inW(y + a + bb): continue
            Op += np.outer(ga, k[abs(bb)] * u[y + bb] * (E[y + bb] - E[y]))
Op = (Op + Op.T) / 2
N = b['N']; K = N - np.diag(np.diag(N))
Ef = np.diag((K * np.outer(u, u)).sum(1)) - K * np.outer(u, u)        # E(h) = h^T Ef h
Pp = np.zeros((M, h)); Pm = np.zeros((M, h))
for j in range(h): Pp[h + j, j] = Pp[h - 1 - j, j] = 1.0; Pm[h + j, j] = 1.0; Pm[h - 1 - j, j] = -1.0
A = Pm.T @ Op @ Pm - Pp.T @ Op @ Pp
B = Pm.T @ Ef @ Pm - Pp.T @ Ef @ Pp
J = np.eye(M)[::-1]; U = np.diag(u)
Hk = (U @ K @ U @ J)[h:, h:]
cos = lambda X, Y: np.sum(X * Y) / np.linalg.norm(X) / np.linalg.norm(Y)
print("%s L=%g M=%d" % (kind, L, M))
print(" check B = 4 Hankel(u eta): |B - 4 Hk| = %.1e" % np.abs(B - 4 * Hk).max())
eA = np.linalg.eigvalsh(A); eB = np.linalg.eigvalsh(B)
print(" cos(A, B) = %.4f ; A: #neg %d of %d (min %.4f max %.4f) ; B: #neg %d of %d (min %.4f max %.4f)" % (cos(A, B), (eA < -1e-10).sum(), h, eA[0], eA[-1], (eB < -1e-10).sum(), h, eB[0], eB[-1]))
print(" cos(Op on even eta, B) = %.4f ; cos(Op on odd eta, B) = %.4f" % (cos(Pp.T @ Op @ Pp, B), cos(Pm.T @ Op @ Pm, B)))
# edge-mirror Hankel form of K in the Doob frame: C_ij = u_i u_j K(x_i, 2L - x_j) -> index i + j' with reflection at upper edge: site j -> 2M-1-j (outside); kernel k(|i - (2M-1-j)|)
C = np.zeros((M, M))
for i in range(M):
    for j in range(M):
        sep = 2 * M - 1 - j - i
        if 0 < sep < M: C[i, j] = k[sep] * u[i] * u[j]
C = C + J @ C @ J                       # both edges
Cf = np.diag(C.sum(1)) - C             # as a Dirichlet-type form in f:  (1/2) sum C_ij (f_i - f_j)^2
print(" edge-mirror forms vs integrated open-path Op (full M x M, mean-zero part): cos(Op, C as matrix) = %.4f ; cos(Op, Dirichlet form of C) = %.4f" % (cos(Op, C), cos(Op, Cf)))
