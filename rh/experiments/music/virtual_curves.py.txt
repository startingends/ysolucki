# Test (Bet A): does "Euler product with nonnegative integer prime counts" + rationality + functional equation
# force RH for a curve-shaped zeta Z(T) = P(T)/((1-T)(1-qT))?  Exact integer arithmetic (Newton recurrences).
import numpy as np
from sympy import mobius, divisors
D = 120
MU = {d: [(mobius(d//e), e) for e in divisors(d)] for d in range(1, D+1)}
def power_sums(coeffs, n_max):
    # coeffs of x^k - c1 x^{k-1} + c2 x^{k-2} - ... : e = [c1, c2, ...]; Newton's identities, exact ints
    k = len(coeffs); e = coeffs; p = [k]
    for n in range(1, n_max+1):
        s = 0
        for i in range(1, min(n, k)+1):
            s += (-1)**(i-1) * e[i-1] * (p[n-i] if n-i > 0 else (n if i == n else 0) * 0 + (p[0] if n-i == 0 else 0))
        if n <= k: s = sum((-1)**(i-1)*e[i-1]*p[n-i] for i in range(1, n)) + (-1)**(n-1)*n*e[n-1]
        else: s = sum((-1)**(i-1)*e[i-1]*p[n-i] for i in range(1, k+1))
        p.append(s)
    return p
def counts_ok(e, q):
    p = power_sums(e, D)
    N = [None] + [q**n + 1 - p[n] for n in range(1, D+1)]
    for d in range(1, D+1):
        s = sum(m*N[ee] for m, ee in MU[d])
        if s < 0 or s % d: return False, d
    return True, None
res = {'g1': [], 'g2': []}
for q in (2, 3, 4, 5, 7, 8, 9, 11, 13, 16, 25):
    for a in range(-2*q-2, 2*q+3):
        al = np.roots([1, -a, q])
        if max(abs(al)) > np.sqrt(q) + 1e-9:
            ok, d = counts_ok([a, q], q)
            if ok and not (abs(max(abs(al)) - q) < 1e-9): res['g1'].append((q, a, np.round(np.abs(al), 3)))
    for a1 in range(-4*q, 4*q+1):
        for a2 in range(-6*q, 6*q+1):
            al = np.roots([1, -a1, a2, -q*a1, q*q])
            if max(abs(al)) > np.sqrt(q) + 1e-6 and max(abs(al)) < q - 1e-6:
                ok, d = counts_ok([a1, a2, q*a1, q*q], q)
                if ok: res['g2'].append((q, a1, a2, np.round(np.abs(al), 3)))
for k in res:
    print(k, "non-RH, all prime counts a_d >= 0 integral for d <=", D, ":", len(res[k]))
    for r in res[k][:6]: print("  ", r)
