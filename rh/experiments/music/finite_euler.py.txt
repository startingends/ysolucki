# Bet A, locally: at height t the archimedean place behaves like one extra string of log-frequency (1/2)log(t/2pi)
# (Riemann-Siegel). Model: F_y(s) = E_y(s) + chi(s) E_y(1-s), E_y = prod_{p<=y} (1-p^-s)^-1 (a STABLE chord: no
# zeros where the product converges). Question: as the prime chord gets louder (y grows), when do zeros leave the
# line?  Count zeros in a box by the argument principle; count on-line zeros by sign changes of
# g(t) = Re(chi^{-1/2} E_y(1/2+it)).  Off-line = total - on-line.
import sys, mpmath as mp
mp.mp.dps = 20
from sympy import primerange
def chi(s): return mp.pi**(s-0.5)*mp.gamma((1-s)/2)/mp.gamma(s/2)
def make(y):
    P = list(primerange(2, y+1))
    def E(s):
        r = mp.mpf(1)
        for p in P: r /= (1 - mp.power(p, -s))
        return r
    return E
def F(E, s): return E(s) + chi(s)*E(1-s)
def winding(E, s0, s1, n=1200):
    # argument principle on rectangle [s0.real, s1.real] x [s0.imag, s1.imag]
    a, b, c, d = s0.real, s1.real, s0.imag, s1.imag
    pts = ([mp.mpc(a + (b-a)*k/n, c) for k in range(n)] + [mp.mpc(b, c + (d-c)*k/n) for k in range(n)] +
           [mp.mpc(b - (b-a)*k/n, d) for k in range(n)] + [mp.mpc(a, d - (d-c)*k/n) for k in range(n)])
    tot = 0; prev = F(E, pts[0])
    for z in pts[1:] + [pts[0]]:
        cur = F(E, z); tot += mp.arg(cur/prev); prev = cur
    return int(mp.nint(tot/(2*mp.pi)))
def online(E, t0, t1, n=4000):
    cnt = 0; prev = None
    for k in range(n+1):
        t = t0 + (t1-t0)*k/n; s = mp.mpc(0.5, t)
        g = mp.re(mp.expj(mp.siegeltheta(t))*E(s))
        if prev is not None and g*prev < 0: cnt += 1
        prev = g
    return cnt
t0 = float(sys.argv[1]) if len(sys.argv) > 1 else 1000.0
dt = float(sys.argv[2]) if len(sys.argv) > 2 else 30.0
YS = [int(v) for v in sys.argv[3].split(',')] if len(sys.argv) > 3 else [3, 10, 30, 100, 300, 1000]
for y in YS:
    E = make(y)
    tot = winding(E, mp.mpc(0.5-0.45, t0+0.123), mp.mpc(0.5+0.45, t0+dt+0.123))
    on = online(E, t0+0.123, t0+dt+0.123)
    print(f"t0={t0:.0f} y={y:5d} zeros in box={tot:3d} on line={on:3d} off line={tot-on:3d}", flush=True)
