# Wallach-type sets of theta powers. Exact rational arithmetic for rational n.
# usage: python3 wallach.py.txt MODE mmax nlo nhi step   (MODE = sq | tri)
import sys
from fractions import Fraction as F
from math import isqrt
def base(mode, mmax):
    f = [F(0)]*(mmax+1)
    if mode == 'sq':
        f[0] = F(1)
        k = 1
        while k*k <= mmax: f[k*k] = F(2); k += 1
    else:
        k = 0
        while k*(k+1)//2 <= mmax: f[k*(k+1)//2] += 1; k += 1
    return f
def power(f, n, mmax):
    # g = f^n, f[0] = 1:  m g_m = sum_{k=1}^m (n k - (m-k)) f_k g_{m-k}
    nz = [k for k in range(1, mmax+1) if f[k] != 0]
    g = [F(0)]*(mmax+1); g[0] = F(1)
    for m in range(1, mmax+1):
        s = F(0)
        for k in nz:
            if k > m: break
            s += (n*k - (m-k))*f[k]*g[m-k]
        g[m] = s/m
    return g
def dlog(a, mmax):
    # a[1] = 1. returns c with exp_*(c) = a (Dirichlet), via L(m) = a(m) log m - sum_{d|m,1<d<m} L(d) a(m/d), c = L/log
    # exact version: use the derivation D(m) = Omega-free trick: work with "log m" replaced by additive function
    # Exact rational: c(m) = a(m) - (1/Om) * ... is messy; use formula with any completely additive weight:
    # take weight w(m) = total number of prime factors Omega(m) (completely additive, rational).
    Om = [0]*(mmax+1)
    for m in range(2, mmax+1):
        x = m; d = 2; c = 0
        while d*d <= x:
            while x % d == 0: x //= d; c += 1
            d += 1
        if x > 1: c += 1
        Om[m] = c
    L = [F(0)]*(mmax+1)
    divs = [[] for _ in range(mmax+1)]
    for d in range(2, mmax+1):
        for m in range(2*d, mmax+1, d): divs[m].append(d)
    for m in range(2, mmax+1):
        s = a[m]*Om[m]
        for d in divs[m]:
            s -= L[d]*a[m//d]
        L[m] = s
    return [None, F(0)] + [L[m]/Om[m] for m in range(2, mmax+1)]
if __name__ == '__main__':
    mode = sys.argv[1]; mmax = int(sys.argv[2]); lo = F(sys.argv[3]); hi = F(sys.argv[4]); st = F(sys.argv[5])
    what = sys.argv[6] if len(sys.argv) > 6 else 'add'
    f = base(mode, mmax)
    n = lo; good = []
    while n <= hi:
        g = power(f, n, mmax)
        if what == 'add':
            bad = next((m for m in range(1, mmax+1) if g[m] < 0), None)
            val = g[bad] if bad else None
        else:
            a = [None] + [g[m]/g[1] for m in range(1, mmax+1)]
            c = dlog(a, mmax)
            bad = next((m for m in range(2, mmax+1) if c[m] < 0), None)
            val = c[bad] if bad else None
        if bad is None: good.append(n)
        else: print(float(n), 'first negative at m =', bad, 'value', float(val))
        n += st
    print('NO negative up to m =', mmax, 'for n in', [str(x) for x in good])
