import sys, math, time, importlib.util, importlib.machinery

weil = importlib.util.module_from_spec(spec); spec.loader.exec_module(weil)
dps = 120; mp.mp.dps = dps; flint.ctx.prec = int(dps * 3.33) + 40

def to_mp(M):
    R = mp.matrix(M.nrows(), M.ncols())
    for i in range(M.nrows()):
        for j in range(M.ncols()):
            R[i, j] = mp.mpf(M[i, j].mid().str(dps + 10, radius=False))
    return R
def blocks(q, N): return [to_mp(weil.even_block(q, N)), to_mp(weil.odd_block(q, N))]
def arch(L, N, pole): return blocks(weil.build(L, N, kind='zeta', pole=pole, primes=False, arch=True)[0], N)
def atoms(L, N, cn):
    """blocks of  -sum_x c_x x^(-1/2) (shift by log x + adjoint);  cn: position x (any real <= e^(2L)) -> weight"""
    cn = {x: c for x, c in cn.items() if math.log(float(x)) < 2 * L - 1e-12}
    if not cn: return None
    saved = weil.dh_logderiv_coeffs
    weil.dh_logderiv_coeffs = lambda X: dict(cn)
    q = weil.build(L, N, kind='dh', pole=False, primes=True, arch=False)[0]
    weil.dh_logderiv_coeffs = saved
    return blocks(q, N)
def add(A, B, t=1): return A if B is None else [A[b] + mp.mpf(t) * B[b] for b in (0, 1)]
def eigs(M): return sorted([e for b in (0, 1) for e in mp.eigsy(M[b], eigvals_only=True)])
def lam(X):
    return {n: arb(p).log() for n, (_, p) in weil.mangoldt_upto(int(X)).items()}

        Qn = add(arch(L, N, pole=False), atoms(L, N, lam(math.exp(2 * L))))      # arch - primes

elif which == "c":

    def beurling(X):
        out = {}
        for p in (2.1, 3, 5, 7, 11, 13):
            pk = p
            while pk <= X:
                out[pk] = arb(p).log(); pk *= p
        return out

    L, N = 0.5, 30
    A = arch(L, N, pole=True)
    print("(d) one atom at position y with free weight m, L = 0.5: for which y does a feasible m exist?")
    for y in (0.45, 0.55, 0.60, 0.64, 0.66, 0.68, 0.69, math.log(2), 0.70, 0.71, 0.73, 0.76, 0.80, 0.90, 0.98):
        U = atoms(L, N, {math.exp(y): arb(1) * arb(math.exp(y)).sqrt()})        # unit weight m: -m (S + S*)
        best = None
        f = lambda m: min(eigs(add(A, U, m)))
        # maximise the lowest eigenvalue over m by golden section on [0, 3]
        lo, hi = mp.mpf(0), mp.mpf(3); g = (mp.sqrt(5) - 1) / 2
        x1, x2 = hi - g * (hi - lo), lo + g * (hi - lo); f1, f2 = f(x1), f(x2)
        for _ in range(40):
            if f1 < f2: lo, x1, f1 = x1, x2, f2; x2 = lo + g * (hi - lo); f2 = f(x2)
            else: hi, x2, f2 = x2, x1, f1; x1 = hi - g * (hi - lo); f1 = f(x1)
        m = (lo + hi) / 2
        print("  y=%.4f (x=%.3f): best m = %s  max lowest eigenvalue = %s  %s" % (y, math.exp(y), mp.nstr(m, 6), mp.nstr(f(m), 4),
