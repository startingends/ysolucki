"""Stage 1: the five promotions. Fields are filled in the LOCAL entry first (the agent's original text of any
replaced field is preserved as a note in its local log), then promoted, then the status is set.
Cross-references use @name@ placeholders. The five main numbers are predicted from the last main number and each
promotion is checked against its prediction (the lead may be writing the main log at the same time)."""
import json, os, sys
exec(open(os.path.join(os.path.dirname(os.path.abspath(__file__)), "audit_lib.py.txt")).read())
marker = once("stage1")
base = main_last()
M = {"alpha": base + 1, "delta": base + 2, "beta": base + 3, "dstar": base + 4, "eps": base + 5}


def R(s):
    for k, v in M.items():
        s = s.replace("@%s@" % k, "#%d" % v)
    assert "@" not in s, s
    return s


def P(name, key, n, fields, status, why):
    m = promote(key, n, {f: R(t) for f, t in fields.items()}, status, R(why) if why else None)
    assert m == M[name], "main number moved: %s got %d, expected %d (fix cross-references by hand)" % (name, m, M[name])
    json.dump(M, open(os.path.join(HERE, "main_numbers.json"), "w"), indent=1)
