"""V-TZ: Delta for every consecutive pair; band argmin per L.  usage: band.py.txt <zeros.jsonl> <q> <L1,L2,...> [fac=1.55]"""
import sys, json, math
rows = [json.loads(l) for l in open(sys.argv[1]) if l.startswith('{"gamma"')]
q = int(sys.argv[2]); Ls = [float(x) for x in sys.argv[3].split(',')]; fac = float(sys.argv[4]) if len(sys.argv) > 4 and sys.argv[4] != "-v" else 1.55
g = [float(r['gamma']) for r in rows]; zp = [float(r['Zp']) for r in rows]
pairs = []
for i in range(len(g) - 1):
    gm = (g[i] + g[i + 1]) / 2; lg = math.log(q * gm / (2 * math.pi))
    D = -zp[i] * zp[i + 1] / lg ** 2
    s = 2 * math.pi / lg; pairs.append((g[i], g[i + 1], D, (g[i + 1] - g[i]) / s))
for L in Ls:
    lo = 2 * math.pi * math.exp(2 * L) / q; hi = fac * lo
    inb = [p for p in pairs if p[0] > lo and p[1] < hi]
    loose = [p for p in pairs if p[1] > lo and p[0] < hi]   # pairs touching the band
    best = min(inb, key=lambda p: p[2]) if inb else None
    bestl = min(loose, key=lambda p: p[2]) if loose else None
    print(f"L = {L}: band ({lo:.2f}, {hi:.2f}); strict argmin: {best and (round(best[0],3), round(best[1],3), round(best[2],4), round(best[3],3))}; "
          f"touching argmin: {bestl and (round(bestl[0],3), round(bestl[1],3), round(bestl[2],4), round(bestl[3],3))}; n strict = {len(inb)}")
if '-v' in sys.argv:
    for p in pairs: print(f"  {p[0]:9.4f} {p[1]:9.4f}  Delta = {p[2]:.4f}  g = {p[3]:.3f}")
