# minimal size of the reflected Toeplitz matrix [s(|i-j|)]_{0..n} that detects G_d at p = 3 (roots 1, -3^d, -3^-d)
import numpy as np
for d in (0.1, 0.01):
    roots = [(1.0, 1), (-3.0**d, 1), (-3.0**-d, 1)]
    s = lambda k: sum(m * a**k for a, m in roots)
    for n in range(1, 6):
        T = np.array([[s(abs(i - j)) for j in range(n + 1)] for i in range(n + 1)])
        print(f"G_d p=3 d={d}: size {n+1}: min eig {np.linalg.eigvalsh(T)[0]:+.6f}")
