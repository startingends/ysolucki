# SP-reflection round 2: formula checks at z = i sqrt5 (class number 2), against omega.
# (1) cosh d(z, gamma z) = 1 + Nb/10, Nb = (b+5c)^2 + 5(a-d)^2;  (2) Na - Nb = 20 det, Na = (b-5c)^2 + 5(a+d)^2;
# (3) genus formula r_Q(n) = sum_{d|n} chi_-20(d) + sum_{d|n} chi_-4(d) chi_5(n/d), Q = x^2+5y^2, for n coprime to 10;
# (4) orbit counts #{gamma in SL_2(Z): Nb = m} against r_Q(m) r_Q(m+20).
import math, random
from collections import Counter
def kron(D, n):  # Kronecker symbol (D/n) for n >= 1 odd or general via sympy
    from sympy import jacobi_symbol
    from sympy.ntheory import factorint
    r = 1
    for p, e in factorint(n).items():
        if p == 2:
            v = 0 if D % 2 == 0 else (1 if D % 8 in (1, 7) else -1)
        else:
            v = jacobi_symbol(D % p, p) if D % p else 0
        r *= v ** e
    return r
z = complex(0, math.sqrt(5)); random.seed(2); err1 = err2 = 0
for _ in range(300):
    while True:
        a, b, c = [random.randint(-9, 9) for _ in range(3)]
        if a and (1 + b * c) % a == 0: d = (1 + b * c) // a; break
    gz = (a * z + b) / (c * z + d)
    ch = 1 + abs(z - gz) ** 2 / (2 * z.imag * gz.imag)
    Nb = (b + 5 * c) ** 2 + 5 * (a - d) ** 2; Na = (b - 5 * c) ** 2 + 5 * (a + d) ** 2
    err1 = max(err1, abs(ch - (1 + Nb / 10))); err2 = max(err2, abs(Na - Nb - 20))
print("max err cosh formula", err1, " max |Na-Nb-20|", err2)
rQ = Counter()
for x in range(-40, 41):
    for y in range(-20, 21): rQ[x * x + 5 * y * y] += 1
bad = 0
for n in range(1, 200):
    if math.gcd(n, 10) == 1:
        divs = [d for d in range(1, n + 1) if n % d == 0]
        f = sum(kron(-20, d) for d in divs) + sum(kron(-4, d) * kron(5, n // d) for d in divs)
        if rQ[n] != f: bad += 1
print("genus formula r_Q(n) = [1*chi_-20 + chi_-4*chi_5](n), n<200 coprime to 10: failures", bad)
B = 30; cnt = Counter()
for a in range(-B, B + 1):
    for b in range(-B, B + 1):
        for c in range(-B, B + 1):
            if a == 0:
                if b * c != -1: continue
                ds = range(-B, B + 1)
            else:
                if (1 + b * c) % a: continue
                ds = [(1 + b * c) // a]
            for d in ds:
                m = (b + 5 * c) ** 2 + 5 * (a - d) ** 2
                if m <= 60: cnt[m] += 1
print(" m  #gamma  r_Q(m) r_Q(m+20)  product")
for m in range(0, 61):
    if cnt[m]: print(f"{m:3d} {cnt[m]:6d} {rQ[m]:6d} {rQ[m+20]:6d} {rQ[m]*rQ[m+20]:7d}")
