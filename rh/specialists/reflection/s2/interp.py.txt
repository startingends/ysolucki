import math
from sympy import mobius
for N in (50, 100, 200, 400):
    mu = [0] + [int(mobius(n)) for n in range(1, N + 1)]
    sq = [n for n in range(1, N + 1) if mu[n]]
    pairs = []
    for m in sq:
        for n in sq:
            g = math.gcd(m, n); a, b = m // g, n // g
            pairs.append((mu[m] * mu[n] / math.sqrt(m * n), 0.5 * abs(math.log(a / b)), 0.5 * math.log(a * b)))
    vals = []
    for k in range(11):
        th = k / 10
        vals.append(sum(w * math.exp(-((1 - th) * pi + th * pf)) for (w, pi, pf) in pairs))
    mono = all(vals[i + 1] <= vals[i] + 1e-12 for i in range(10))
    print(f"N={N}: Q_0 = {vals[0]:.5f}  Q_0.5 = {vals[5]:.5f}  Q_1 = {vals[10]:.5f}  monotone decreasing: {mono}")
    print("   ", " ".join(f"{v:.4f}" for v in vals))
