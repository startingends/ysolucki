# S2/B5 formula check of the gcd identity (*): for any even function g (Fourier transform of a positive t-measure),
#   sum_{a,b<=N} mu(a)mu(b)(ab)^(-1/2) g(log(a/b))  ==  sum_{(a,b)=1, a,b<=N} g(log(a/b)) mu(a)mu(b)(ab)^(-1/2) S_ab(N/max(a,b)),
#   S_r(x) = sum_{k<=x, (k,r)=1} mu(k)^2/k.   Also: local density of sigma_p >= 0 and its Fourier coefficient at +-1.
import math
from sympy import mobius
N = 80; T = 7.0
g = lambda x: math.exp(-(T * x) ** 2 / 2)       # Gaussian t-window of width T
mu = [0] + [int(mobius(n)) for n in range(1, N + 1)]
lhs = sum(mu[a] * mu[b] / math.sqrt(a * b) * g(math.log(a / b)) for a in range(1, N + 1) for b in range(1, N + 1))
def S(r, x):
    return sum(mu[k] ** 2 / k for k in range(1, int(x) + 1) if math.gcd(k, r) == 1)
rhs = sum(g(math.log(a / b)) * mu[a] * mu[b] / math.sqrt(a * b) * S(a * b, N / max(a, b))
          for a in range(1, N + 1) for b in range(1, N + 1) if math.gcd(a, b) == 1)
torus = sum(mu[k] ** 2 / k for k in range(1, N + 1))
print(f"N={N}, T={T}: LHS {lhs:.12f}  RHS {rhs:.12f}  diff {lhs-rhs:.2e}   torus value (g = delta) {torus:.6f}")
for p in (2, 3, 5, 7):
    mn = min((p / (p + 1)) * abs(1 - p ** -0.5 * complex(math.cos(th), math.sin(th))) ** 2 for th in [i * 2 * math.pi / 1000 for i in range(1000)])
    c1 = sum((p / (p + 1)) * abs(1 - p ** -0.5 * complex(math.cos(th), math.sin(th))) ** 2 * math.cos(th) for th in [i * 2 * math.pi / 4000 for i in range(4000)]) / 4000
    print(f"p={p}: min local density {mn:.5f} (>0), Fourier coeff at 1: {c1:.6f} vs -sqrt(p)/(p+1) = {-math.sqrt(p)/(p+1):.6f}")
