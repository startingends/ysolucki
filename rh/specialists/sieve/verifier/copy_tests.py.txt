def kronecker(a, n):
    """Kronecker symbol (a/n) for n >= 1."""
    if n == 1:
        return 1
    res = 1
    # factor out 2s from n
    while n % 2 == 0:
        n //= 2
        if a % 2 == 0:
            return 0
        if a % 8 in (3, 5):
            res = -res
    # now n odd: Jacobi symbol (a/n)
    a %= n
    while a:
        while a % 2 == 0:
            a //= 2
            if n % 8 in (3, 5):
                res = -res
        a, n = n, a
        if a % 4 == 3 and n % 4 == 3:
            res = -res
        a %= n
    return res if n == 1 else 0

# ---------------- T2: all Dirichlet characters mod q <= Q ----------------
print("== T2 Dirichlet family: all characters mod q, q<=Q, on n<=D ==")
def phi(n):
    r, m, p = n, n, 2
    while p * p <= m:
        if m % p == 0:
            while m % p == 0:
                m //= p
            r -= r // p
        p += 1
    if m > 1:
        r -= r // m
    return r

def subgroup_size(q, gens):
    if q == 1:
        return 1
    seen = {1}
    frontier = [1]
    gens = [g % q for g in gens]
    while frontier:
        nf = []
        for x in frontier:
            for g in gens:
                y = x * g % q
                if y not in seen:
                    seen.add(y)
                    nf.append(y)
        frontier = nf
    return len(seen)

for Q in (100, 200, 400):

def ideal_count_quadratic(disc, X):
    """I(X) = #{ideals of norm <= X} in Q(sqrt disc), via a(n) = sum_{d|n} (disc/d)."""
    tot = 0

# ---------------- T4: squarefull counts, independent method ----------------
print("== T4 squarefull counts by smallest-prime-factor sieve ==")
N = 10**7
spf = list(range(N + 1))
for p in range(2, isqrt(N) + 1):
    if spf[p] == p:
        for k in range(p * p, N + 1, p):
            if spf[k] == k:
                spf[k] = p
def is_squarefull(n):
    while n > 1:
        p = spf[n]
        e = 0
        while n % p == 0:
            n //= p
            e += 1
        if e == 1:
            return False
    return True
cnt = 0
checkpoints = {10**4, 10**5, 10**6, 10**7}
worst = 0.0
for n in range(1, N + 1):
    if is_squarefull(n):
        cnt += 1
        worst = max(worst, cnt / math.sqrt(n))
