# V-CUB: SP-sieve's reconstructed system (rh/specialists/sieve/slope_system.py.txt, loaded unchanged) with the SIGNAL term
# replaced by the one the two-variable cubic family actually has.
#   Model's signal term (c = 1/3, one-variable Poisson):  h/3
#   Two-variable family (two Poisson summations, each normalised by N^(-1/2), cube pairs of the dual box):  h/6 - l_x/6
#     (= model - (1 + l)/6 = sextic - l_x/6; derivation in VERDICT.md sec. 2)
# Usage: python3 corrected_signal.py.txt <clow> <alpha> <rho> <cubic:0/1> [sig: two|model]
#   clow = copy density used in the low estimate's exceptional-frequency term clow*b/2 (chain: 1/3)
import sys, numpy as np
src = open('/home/user/ysolucki/rh/specialists/sieve/slope_system.py.txt').read().split("if __name__ == '__main__':")[0]
ns = {}; exec(src, ns)
SIG = 'two'
def sigterm(lx, h):
    return h/6 - lx/6 if SIG == 'two' else h/3
def Emax(sigma0, lx, b, csig, alpha, rho, delta0=1/50, nd=401, nx=21, use_dyn_kappa=True, tmax=1.5, need=None):
    ell = 1 - 2*lx - b; h = 1 - lx + ell; ly = lx + b
    if ell < 0 or b < 0 or lx <= 0: return 1e9
    kappa = 0.75
    sixk = 6*kappa; cap = ell/h
    delta = np.linspace(delta0, alpha, nd); x = np.linspace(0, 0.5, nx)
    R = ns['Rstar'](delta, x, alpha, rho, sixk, cap, tmax=tmax)
    D = delta[:,None]; X = x[None,:]
    S = sigterm(lx, h)
    E = (1+D)*(1-ly)/2 - sigma0 - S - ell/2 + X*D*ell + h*R + h*D/2
    Ef = (1+delta0)*(1-ly)/2 - sigma0 - S - ell/2 + 0.5*delta0*ell + h + h*delta0/2
    return max(E.max(), Ef)
CUB = True
def sigma_low(lx, b, clow):
    ell = 1 - 2*lx - b; h = 1 - lx + ell
    base = 1 - sigterm(lx, h) + clow*b/2
    if CUB:
        M = 2*lx + b
        base += 0.5*max(0.0, (2/3)*(3*M - 1) - max(M, 2*M - 1))
    return base
ns['Emax'] = Emax; ns['sigma_low'] = sigma_low
if __name__ == '__main__':
    clow, alpha, rho = float(sys.argv[1]), float(sys.argv[2]), float(sys.argv[3])
    CUB = sys.argv[4] == '1'; ns['CUB'] = CUB
    if len(sys.argv) > 5: SIG = sys.argv[5]
    globals()['CUB'] = CUB
    starts = [(17/48, 1/8), (0.3, 0.05), (0.42, 0.1), (0.22, 0.15), (0.1, 0.0), (0.05, 0.3)]
    s, g = ns['optimise'](clow, alpha, rho, 1.5, starts=starts)
    print(f"SIG={SIG} clow={clow:.4f} alpha={alpha:.4f} rho={rho:.4f} cubic_low={CUB}: best sigma0 = {s:.6f} at (l_x, b) = "
          f"({g[0]:.5f}, {g[1]:.5f}); low line there {sigma_low(g[0], g[1], clow):.6f}")
    # also at the chain's C1 optimum and the paper geometry
    for (lx, b) in [(0.37917, 0.05510), (17/48, 1/8)]:
        print(f"  at (l_x, b) = ({lx:.5f}, {b:.5f}): sigma_best = {ns['sigma_best_geom'](lx, b, clow, alpha, rho, 1.5, fine=True):.6f}, "
              f"low line {sigma_low(lx, b, clow):.6f}")
