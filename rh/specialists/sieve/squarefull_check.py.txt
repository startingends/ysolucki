# Check the counting core of COPY_BARRIER.md: #{n <= X squarefull} <= zeta(3/2) X^(1/2) (bound via n = a^2 b^3, b squarefree),
# against the asymptotic (zeta(3/2)/zeta(3)) X^(1/2); and #{m-th powers <= X} = floor(X^(1/m)).
import math
def squarefull_count(X):
    # n = a^2 b^3 with b squarefree, unique representation
    sf = bytearray([1])*(int(round(X**(1/3)))+2)
    B = len(sf)-1
    for p in range(2, int(B**0.5)+1):
        for k in range(p*p, B+1, p*p): sf[k] = 0
    c = 0
    b = 1
    while b**3 <= X:
        if sf[b]:
            c += math.isqrt(X // b**3)
        b += 1
    return c
z32 = 2.6123753486854883; z3 = 1.2020569031595942
for X in (10**4, 10**6, 10**8, 10**10, 10**12):
    c = squarefull_count(X)
    print(f"X=1e{round(math.log10(X))}: squarefull {c}, /sqrt(X) = {c/math.sqrt(X):.5f}, asymptotic const zeta(3/2)/zeta(3) = {z32/z3:.5f}, bound zeta(3/2) = {z32:.5f}")
