# unitary-side check of NOTES D12 (c): (1/(pi w)) int |fhat|^2 sin^2(arg xi_S(1/2 + w - ix)) dx  ->  sum_gamma |fhat(gamma)|^2
import mpmath as mp
mp.mp.dps = 20
s0 = mp.mpf('0.12')
fhat2 = lambda x: (s0*mp.sqrt(2*mp.pi))**2*mp.exp(-s0*s0*x*x)
def xiS(s): return s*(s-1)*mp.pi**(-s/2)*mp.gamma(s/2)*mp.zeta(s)
gam = [mp.im(mp.zetazero(k)) for k in range(1, 18)]
zsum = 2*sum(fhat2(g) for g in gam)
print("zero sum (17 zeros each sign, |gamma| < 70):", mp.nstr(zsum, 12))
for w in [mp.mpf('0.2'), mp.mpf('0.1'), mp.mpf('0.05'), mp.mpf('0.02'), mp.mpf('0.01')]:
    def integrand(x):
        v = xiS(mp.mpf(1)/2 + w - 1j*x)
        return fhat2(x)*(mp.im(v)/abs(v))**2          # sin^2 of the argument
    # break points at the zeros (the integrand peaks there with width w)
    pts = [mp.mpf(0)]
    for g in gam:
        if g < 62: pts += [g - 5*w, g, g + 5*w]
    pts.append(mp.mpf(62))
    val = 2*mp.quad(integrand, pts)/(mp.pi*w)
    print("w =", mp.nstr(w, 3), "  (1/(pi w)) int =", mp.nstr(val, 12), "  ratio to zero sum =", mp.nstr(val/zsum, 8))
