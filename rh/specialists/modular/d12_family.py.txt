def qw(n, w):
    r, m, p = mp.mpf(n)**w, n, 2
    while p*p <= m:
        if m % p == 0:
            r *= (1 - mp.mpf(p)**(-2*w))
            while m % p == 0: m //= p
        p += 1
    if m > 1: r *= (1 - mp.mpf(m)**(-2*w))
    return r
