"""Local-zeros refinement of the two-zero model (TWO_ZERO.md section 4).
Q~(F) = ||F||^2 - s sum_{lattice points within W of c} |F(p)|^2 + s sum_{actual on-line zeros within W} |F(gamma)|^2
        + s sum_{off-line points z = gamma + i b within W} 2 Re F(z) conj F(conj z),
on PW_L (||F||^2 = int |F|^2 dt), lattice c + s(Z + 1/2) beyond W (tight frame part). Exact finite eigenproblem:
spectrum = {1} u {1 + eig(Gamma^(1/2) H Gamma^(1/2))}, Gamma_ij = s K(z_i, z_j), K(z, w) = sin(L(z - conj w))/(pi(z - conj w)).
With W = 0 (no actual zeros) and a single pair this is the closed form of model.py.txt (checked below)."""

def Kf(z, w, L):
    d = z - complex(w).conjugate()
    return L / math.pi if abs(d) < 1e-14 else cmath.sin(L * d) / (math.pi * d)

def local_model(L, q, c, onl, offl, W, s=None):
    """returns (rho = min eigenvalue of Q~, shares dict for the minimiser, s, r)"""
    if s is None: s = 2 * math.pi / math.log(q * c / (2 * math.pi))
    r = math.pi / (L * s)
    pts = []; H = []
    K = int(W / s) + 2
    lat = [c + s * (k + 0.5) for k in range(-K, K) if abs(s * (k + 0.5)) < W]
    for p in lat: pts.append(p); H.append(('lat', -1.0))
    for g in onl:
        if abs(g - c) < W: pts.append(g); H.append(('on', 1.0))
    offidx = []
    for b, g in offl:
        if abs(g - c) < W:
            offidx.append((len(pts), len(pts) + 1)); pts.append(complex(g, b)); pts.append(complex(g, -b)); H.append(('off', 0)); H.append(('off', 0))
    n = len(pts)
    Hm = np.zeros((n, n), dtype=complex)
    for i, (kind, e) in enumerate(H):
        if kind != 'off': Hm[i, i] = e
    for i, j in offidx: Hm[i, j] = 1; Hm[j, i] = 1
    G = np.array([[s * Kf(pts[i], pts[j], L) for j in range(n)] for i in range(n)])
    G = (G + G.conj().T) / 2
    w, U = np.linalg.eigh(G); w = np.clip(w, 0, None)
    G12 = (U * np.sqrt(w)) @ U.conj().T
    A = G12 @ Hm @ G12; A = (A + A.conj().T) / 2
    ev, Y = np.linalg.eigh(A)
    lam = ev[0]; rho_ = min(1.0, 1 + lam)
    y = Y[:, 0]; v = G12 @ y            # evaluations (scaled by sqrt(s)); ||F||^2 = |y|^2 = 1
    qtot = 1 + lam
    sh = {'lattice rest': (1 - sum(abs(v[i]) ** 2 for i, (k, e) in enumerate(H) if k == 'lat')).real}
    for i, (k, e) in enumerate(H):
        if k == 'on': sh[f'on {pts[i].real if isinstance(pts[i], complex) else pts[i]:.3f}'] = abs(v[i]) ** 2
    for i, j in offidx: sh[f'off {pts[i].real:.2f}'] = (2 * (v[i] * np.conj(v[j])).real)
    return rho_, {k: x / qtot for k, x in sh.items()}, s, r

def show(name, L, q, c, onl, offl, W):
    rh, sh, s, r = local_model(L, q, c, onl, offl, W)
    big = sorted(sh.items(), key=lambda kv: -abs(kv[1]))[:5]
    print(f"  {name}: L = {L}, W = {W}: s = {s:.4f}, r = {r:.4f}, rho = {rh:.5f}; shares of Q at the minimiser: " + ", ".join(f"{k}: {x:.3f}" for k, x in big), flush=True)
    return rh
